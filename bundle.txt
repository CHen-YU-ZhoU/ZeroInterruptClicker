std::string jsonEscape(const std::string& input) {
    std::string out;
    out.reserve(input.size() + 8);
    for (char c : input) {
        if (c == '"' || c == '\\') {
            out.push_back('\\');
            out.push_back(c);
        } else if (c == '\n') {
            out += "\\n";
        } else {
            out.push_back(c);
        }
    }
    return out;
}

constexpr size_t kBitFieldBytes(size_t maxBit) {
    return (maxBit + 7) / 8 + 1;
}

template <size_t N>
bool testBit(const std::array<uint8_t, N>& bits, size_t bit) {
    const size_t byteIdx = bit / 8;
    if (byteIdx >= bits.size()) {
        return false;
    }
    return (bits[byteIdx] & (1u << (bit % 8))) != 0;
}

int64_t TouchMuxEngine::nowMonotonicNs() {
    struct timespec ts {};
    clock_gettime(CLOCK_MONOTONIC, &ts);
    return static_cast<int64_t>(ts.tv_sec) * 1000000000LL + static_cast<int64_t>(ts.tv_nsec);
}

bool TouchMuxEngine::findBestTouchscreen(DeviceDesc& outDesc, int& outErrno) {
    outErrno = ENODEV;
    DIR* dir = opendir("/dev/input");
    if (!dir) {
        outErrno = errno;
        return false;
    }

    int bestScore = -1;
    DeviceDesc best{};

    struct dirent* ent = nullptr;
    while ((ent = readdir(dir)) != nullptr) {
        if (std::strncmp(ent->d_name, "event", 5) != 0) {
            continue;
        }
        std::string path = std::string("/dev/input/") + ent->d_name;
        int fd = open(path.c_str(), O_RDONLY | O_NONBLOCK | O_CLOEXEC);
        if (fd < 0) {
            if (errno == EACCES || errno == EPERM) {
                outErrno = errno;
            }
            continue;
        }

        char nameBuf[256] = {};
        if (ioctl(fd, EVIOCGNAME(sizeof(nameBuf) - 1), nameBuf) < 0) {
            std::strncpy(nameBuf, "unknown", sizeof(nameBuf) - 1);
        }

        if (std::strstr(nameBuf, "ZeroInterrupt_TouchMux") != nullptr) {
            close(fd);
            continue;
        }

        std::array<uint8_t, kBitFieldBytes(EV_MAX)> evBits{};
        std::array<uint8_t, kBitFieldBytes(ABS_MAX)> absBits{};
        std::array<uint8_t, kBitFieldBytes(KEY_MAX)> keyBits{};
        std::array<uint8_t, kBitFieldBytes(INPUT_PROP_MAX)> propBits{};

        if (ioctl(fd, EVIOCGBIT(0, evBits.size()), evBits.data()) < 0) {
            close(fd);
            continue;
        }
        if (!testBit(evBits, EV_ABS)) {
            close(fd);
            continue;
        }

        ioctl(fd, EVIOCGBIT(EV_ABS, absBits.size()), absBits.data());
        if (!testBit(absBits, ABS_MT_POSITION_X) || !testBit(absBits, ABS_MT_POSITION_Y)) {
            close(fd);
            continue;
        }

        if (testBit(evBits, EV_KEY)) {
            ioctl(fd, EVIOCGBIT(EV_KEY, keyBits.size()), keyBits.data());
        }
        ioctl(fd, EVIOCGPROP(propBits.size()), propBits.data());

        int score = 10;
        if (testBit(propBits, INPUT_PROP_DIRECT)) score += 100;
        if (testBit(keyBits, BTN_TOUCH)) score += 50;
        if (testBit(absBits, ABS_MT_SLOT)) score += 30;
        if (testBit(absBits, ABS_MT_TRACKING_ID)) score += 20;
        if (testBit(absBits, ABS_MT_TOUCH_MAJOR)) score += 5;
        if (testBit(absBits, ABS_MT_PRESSURE)) score += 5;

        if (score > bestScore) {
            DeviceDesc candidate;
            candidate.path = path;
            candidate.name = nameBuf;
            struct input_id idInfo {};
            if (ioctl(fd, EVIOCGID, &idInfo) == 0) {
                candidate.bustype = idInfo.bustype;
                candidate.vendor = idInfo.vendor;
                candidate.product = idInfo.product;
                candidate.version = idInfo.version;
            }
            ioctl(fd, EVIOCGABS(ABS_MT_POSITION_X), &candidate.absX);
            ioctl(fd, EVIOCGABS(ABS_MT_POSITION_Y), &candidate.absY);

            candidate.isTypeB = testBit(absBits, ABS_MT_SLOT);
            if (candidate.isTypeB) {
                ioctl(fd, EVIOCGABS(ABS_MT_SLOT), &candidate.absSlot);
                int maxSlot = candidate.absSlot.maximum;
                candidate.slotCount = std::clamp(maxSlot + 1, 1, kMaxPhysSlots);
            } else {
                candidate.slotCount = 10;
            }
            if (testBit(absBits, ABS_MT_TRACKING_ID)) {
                ioctl(fd, EVIOCGABS(ABS_MT_TRACKING_ID), &candidate.absTracking);
            }
            candidate.hasTouchMajor = testBit(absBits, ABS_MT_TOUCH_MAJOR);
            if (candidate.hasTouchMajor) {
                ioctl(fd, EVIOCGABS(ABS_MT_TOUCH_MAJOR), &candidate.absTouchMajor);
            }
            candidate.hasWidthMajor = testBit(absBits, ABS_MT_WIDTH_MAJOR);
            if (candidate.hasWidthMajor) {
                ioctl(fd, EVIOCGABS(ABS_MT_WIDTH_MAJOR), &candidate.absWidthMajor);
            }
            candidate.hasPressure = testBit(absBits, ABS_MT_PRESSURE);
            if (candidate.hasPressure) {
                ioctl(fd, EVIOCGABS(ABS_MT_PRESSURE), &candidate.absPressure);
            }
            candidate.hasBtnToolFinger = testBit(keyBits, BTN_TOOL_FINGER);

            if (candidate.absX.maximum > candidate.absX.minimum &&
                candidate.absY.maximum > candidate.absY.minimum) {
                bestScore = score;
                best = std::move(candidate);
            }
        }
        close(fd);
    }
    closedir(dir);

    if (bestScore < 0) {
        return false;
    }
    outDesc = std::move(best);
    outErrno = 0;
    return true;
}

ProbeResult TouchMuxEngine::probeDevices() {
    ProbeResult result;
    int ufd = open("/dev/uinput", O_WRONLY | O_NONBLOCK | O_CLOEXEC);
    if (ufd >= 0) {
        result.uinputWritable = true;
        result.uinputErrno = 0;
        close(ufd);
    } else {
        result.uinputWritable = false;
        result.uinputErrno = errno;
    }

    DeviceDesc desc;
    int touchErr = 0;
    if (findBestTouchscreen(desc, touchErr)) {
        result.touchDevicePath = desc.path;
        result.touchDeviceName = desc.name;
        result.touchDeviceReadable = true;
        result.touchDeviceErrno = 0;
        result.physSlotCount = desc.slotCount;
        result.isTypeB = desc.isTypeB;
        result.absMinX = desc.absX.minimum;
        result.absMaxX = desc.absX.maximum;
        result.absMinY = desc.absY.minimum;
        result.absMaxY = desc.absY.maximum;
    } else {
        result.touchDeviceReadable = false;
        result.touchDeviceErrno = touchErr;
    }
    return result;
}

bool TouchMuxEngine::setupUinputDevice(const DeviceDesc& desc, int& outErrno) {
    uinputFd_ = open("/dev/uinput", O_WRONLY | O_NONBLOCK | O_CLOEXEC);
    if (uinputFd_ < 0) {
        outErrno = errno;
        lastError_ = formatErrno("open(/dev/uinput) failed", outErrno);
        return false;
    }

    auto setBitOrFail = [this, &outErrno](unsigned long req, int val, const char* label) -> bool {
        if (ioctl(uinputFd_, req, val) < 0) {
            outErrno = errno;
            lastError_ = formatErrno(label, outErrno);
            return false;
        }
        return true;
    };

    if (!setBitOrFail(UI_SET_EVBIT, EV_SYN, "UI_SET_EVBIT(EV_SYN)")) return false;
    if (!setBitOrFail(UI_SET_EVBIT, EV_KEY, "UI_SET_EVBIT(EV_KEY)")) return false;
    if (!setBitOrFail(UI_SET_EVBIT, EV_ABS, "UI_SET_EVBIT(EV_ABS)")) return false;

    if (!setBitOrFail(UI_SET_KEYBIT, BTN_TOUCH, "UI_SET_KEYBIT(BTN_TOUCH)")) return false;
    if (desc.hasBtnToolFinger) {
        ioctl(uinputFd_, UI_SET_KEYBIT, BTN_TOOL_FINGER);
    }

    if (ioctl(uinputFd_, UI_SET_PROPBIT, INPUT_PROP_DIRECT) < 0) {
        LOGW("UI_SET_PROPBIT(INPUT_PROP_DIRECT) warning: %s", std::strerror(errno));
    }

    if (!setBitOrFail(UI_SET_ABSBIT, ABS_MT_SLOT, "UI_SET_ABSBIT(ABS_MT_SLOT)")) return false;
    if (!setBitOrFail(UI_SET_ABSBIT, ABS_MT_TRACKING_ID, "UI_SET_ABSBIT(ABS_MT_TRACKING_ID)")) return false;
    if (!setBitOrFail(UI_SET_ABSBIT, ABS_MT_POSITION_X, "UI_SET_ABSBIT(ABS_MT_POSITION_X)")) return false;
    if (!setBitOrFail(UI_SET_ABSBIT, ABS_MT_POSITION_Y, "UI_SET_ABSBIT(ABS_MT_POSITION_Y)")) return false;

    if (desc.hasTouchMajor) ioctl(uinputFd_, UI_SET_ABSBIT, ABS_MT_TOUCH_MAJOR);
    if (desc.hasWidthMajor) ioctl(uinputFd_, UI_SET_ABSBIT, ABS_MT_WIDTH_MAJOR);
    if (desc.hasPressure) ioctl(uinputFd_, UI_SET_ABSBIT, ABS_MT_PRESSURE);

    bool configuredModern = false;
#if defined(UI_DEV_SETUP) && defined(UI_ABS_SETUP)
    {
        auto setupAbs = [this](uint16_t code, const struct input_absinfo& info) -> bool {
            struct uinput_abs_setup absSetup {};
            absSetup.code = code;
            absSetup.absinfo = info;
            return ioctl(uinputFd_, UI_ABS_SETUP, &absSetup) == 0;
        };

        struct input_absinfo slotInfo {};
        slotInfo.minimum = 0;
        slotInfo.maximum = kMaxUinputSlots - 1;

        struct input_absinfo trackInfo {};
        trackInfo.minimum = 0;
        trackInfo.maximum = 65535;

        bool absOk = setupAbs(ABS_MT_SLOT, slotInfo) &&
                     setupAbs(ABS_MT_TRACKING_ID, trackInfo) &&
                     setupAbs(ABS_MT_POSITION_X, desc.absX) &&
                     setupAbs(ABS_MT_POSITION_Y, desc.absY);

        if (absOk && desc.hasTouchMajor) setupAbs(ABS_MT_TOUCH_MAJOR, desc.absTouchMajor);
        if (absOk && desc.hasWidthMajor) setupAbs(ABS_MT_WIDTH_MAJOR, desc.absWidthMajor);
        if (absOk && desc.hasPressure) setupAbs(ABS_MT_PRESSURE, desc.absPressure);

        if (absOk) {
            struct uinput_setup devSetup {};
            std::snprintf(devSetup.name, UINPUT_MAX_NAME_SIZE, "ZeroInterrupt_TouchMux");
            devSetup.id.bustype = BUS_VIRTUAL;
            devSetup.id.vendor = desc.vendor != 0 ? desc.vendor : 0x18D1;
            devSetup.id.product = desc.product != 0 ? desc.product : 0x4E11;
            devSetup.id.version = desc.version != 0 ? desc.version : 1;
            if (ioctl(uinputFd_, UI_DEV_SETUP, &devSetup) == 0) {
                configuredModern = true;
            }
        }
    }
#endif

    if (!configuredModern) {
        struct uinput_user_dev uidev {};
        std::snprintf(uidev.name, UINPUT_MAX_NAME_SIZE, "ZeroInterrupt_TouchMux");
        uidev.id.bustype = BUS_VIRTUAL;
        uidev.id.vendor = desc.vendor != 0 ? desc.vendor : 0x18D1;
        uidev.id.product = desc.product != 0 ? desc.product : 0x4E11;
        uidev.id.version = desc.version != 0 ? desc.version : 1;

        uidev.absmin[ABS_MT_SLOT] = 0;
        uidev.absmax[ABS_MT_SLOT] = kMaxUinputSlots - 1;
        uidev.absmin[ABS_MT_TRACKING_ID] = 0;
        uidev.absmax[ABS_MT_TRACKING_ID] = 65535;

        uidev.absmin[ABS_MT_POSITION_X] = desc.absX.minimum;
        uidev.absmax[ABS_MT_POSITION_X] = desc.absX.maximum;
        uidev.absfuzz[ABS_MT_POSITION_X] = desc.absX.fuzz;
        uidev.absflat[ABS_MT_POSITION_X] = desc.absX.flat;

        uidev.absmin[ABS_MT_POSITION_Y] = desc.absY.minimum;
        uidev.absmax[ABS_MT_POSITION_Y] = desc.absY.maximum;
        uidev.absfuzz[ABS_MT_POSITION_Y] = desc.absY.fuzz;
        uidev.absflat[ABS_MT_POSITION_Y] = desc.absY.flat;

        if (desc.hasTouchMajor) {
            uidev.absmin[ABS_MT_TOUCH_MAJOR] = desc.absTouchMajor.minimum;
            uidev.absmax[ABS_MT_TOUCH_MAJOR] = desc.absTouchMajor.maximum;
            uidev.absfuzz[ABS_MT_TOUCH_MAJOR] = desc.absTouchMajor.fuzz;
            uidev.absflat[ABS_MT_TOUCH_MAJOR] = desc.absTouchMajor.flat;
        }
        if (desc.hasWidthMajor) {
            uidev.absmin[ABS_MT_WIDTH_MAJOR] = desc.absWidthMajor.minimum;
            uidev.absmax[ABS_MT_WIDTH_MAJOR] = desc.absWidthMajor.maximum;
        }
        if (desc.hasPressure) {
            uidev.absmin[ABS_MT_PRESSURE] = desc.absPressure.minimum;
            uidev.absmax[ABS_MT_PRESSURE] = desc.absPressure.maximum;
            uidev.absfuzz[ABS_MT_PRESSURE] = desc.absPressure.fuzz;
            uidev.absflat[ABS_MT_PRESSURE] = desc.absPressure.flat;
        }

        if (write(uinputFd_, &uidev, sizeof(uidev)) != static_cast<ssize_t>(sizeof(uidev))) {
            outErrno = errno;
            lastError_ = formatErrno("write(uinput_user_dev) failed", outErrno);
            return false;
        }
    }

    if (ioctl(uinputFd_, UI_DEV_CREATE) < 0) {
        outErrno = errno;
        lastError_ = formatErrno("ioctl(UI_DEV_CREATE) failed", outErrno);
        return false;
    }

    outErrno = 0;
    return true;
}

void TouchMuxEngine::closeAllFds() {
    if (epollFd_ >= 0) {
        close(epollFd_);
        epollFd_ = -1;
    }
    if (timerFd_ >= 0) {
        close(timerFd_);
        timerFd_ = -1;
    }
    if (wakeFd_ >= 0) {
        close(wakeFd_);
        wakeFd_ = -1;
    }
    if (physFd_ >= 0) {
        close(physFd_);
        physFd_ = -1;
    }
    if (uinputFd_ >= 0) {
        close(uinputFd_);
        uinputFd_ = -1;
    }
}

void TouchMuxEngine::signalWake() {
    if (wakeFd_ >= 0) {
        uint64_t val = 1;
        ssize_t ignored = write(wakeFd_, &val, sizeof(val));
        (void)ignored;
    }
}

int TouchMuxEngine::initAndGrab(int displayW, int displayH, int rotation, int selfCheckTimeoutMs) {
    if (threadRunning_.load()) {
        {
            std::lock_guard<std::mutex> lock(stateMutex_);
            cfgDisplayW_ = std::max(1, displayW);
            cfgDisplayH_ = std::max(1, displayH);
            cfgRotation_ = rotation;
            if (selfCheckTimeoutMs > 0) {
                cfgSelfCheckDeadlineNs_ = nowMonotonicNs() + static_cast<int64_t>(selfCheckTimeoutMs) * 1000000LL;
                selfCheckPending_.store(true);
                selfCheckRolledBack_.store(false);
                engineState_.store(static_cast<int32_t>(EngineState::SELF_CHECK_PENDING));
            }
        }
        signalWake();
        return 0;
    }

    if (workerThread_.joinable()) {
        workerThread_.join();
    }

    DeviceDesc desc;
    int findErr = 0;
    if (!findBestTouchscreen(desc, findErr)) {
        lastError_ = formatErrno("findBestTouchscreen failed", findErr);
        engineState_.store(static_cast<int32_t>(EngineState::ERROR_STATE));
        return -findErr;
    }

    physFd_ = open(desc.path.c_str(), O_RDONLY | O_NONBLOCK | O_CLOEXEC);
    if (physFd_ < 0) {
        int err = errno;
        lastError_ = formatErrno("open physical touchscreen failed", err);
        engineState_.store(static_cast<int32_t>(EngineState::ERROR_STATE));
        return -err;
    }

    int uinputErr = 0;
    if (!setupUinputDevice(desc, uinputErr)) {
        closeAllFds();
        engineState_.store(static_cast<int32_t>(EngineState::ERROR_STATE));
        return -uinputErr;
    }

    if (ioctl(physFd_, EVIOCGRAB, 1) < 0) {
        int err = errno;
        lastError_ = formatErrno("ioctl(EVIOCGRAB, 1) failed", err);
        ioctl(uinputFd_, UI_DEV_DESTROY);
        closeAllFds();
        engineState_.store(static_cast<int32_t>(EngineState::ERROR_STATE));
        return -err;
    }

    epollFd_ = epoll_create1(EPOLL_CLOEXEC);
    timerFd_ = timerfd_create(CLOCK_MONOTONIC, TFD_NONBLOCK | TFD_CLOEXEC);
    wakeFd_ = eventfd(0, EFD_NONBLOCK | EFD_CLOEXEC);
    if (epollFd_ < 0 || timerFd_ < 0 || wakeFd_ < 0) {
        int err = errno;
        lastError_ = formatErrno("epoll/timerfd/eventfd creation failed", err);
        ioctl(physFd_, EVIOCGRAB, 0);
        ioctl(uinputFd_, UI_DEV_DESTROY);
        closeAllFds();
        engineState_.store(static_cast<int32_t>(EngineState::ERROR_STATE));
        return -err;
    }

    auto addEpoll = [this](int fd) -> bool {
        struct epoll_event ev {};
        ev.events = EPOLLIN;
        ev.data.fd = fd;
        return epoll_ctl(epollFd_, EPOLL_CTL_ADD, fd, &ev) == 0;
    };

    if (!addEpoll(physFd_) || !addEpoll(timerFd_) || !addEpoll(wakeFd_)) {
        int err = errno;
        lastError_ = formatErrno("epoll_ctl ADD failed", err);
        ioctl(physFd_, EVIOCGRAB, 0);
        ioctl(uinputFd_, UI_DEV_DESTROY);
        closeAllFds();
        engineState_.store(static_cast<int32_t>(EngineState::ERROR_STATE));
        return -err;
    }

    {
        std::lock_guard<std::mutex> lock(stateMutex_);
        deviceDesc_ = desc;
        lastError_.clear();
        cfgDisplayW_ = std::max(1, displayW);
        cfgDisplayH_ = std::max(1, displayH);
        cfgRotation_ = rotation;
        cfgClickingRunning_ = false;
        cfgTargets_.clear();
        cfgCancelSelfCheckRequested_ = false;
        cfgStopThreadRequested_ = false;
        if (selfCheckTimeoutMs > 0) {
            cfgSelfCheckDeadlineNs_ = nowMonotonicNs() + static_cast<int64_t>(selfCheckTimeoutMs) * 1000000LL;
            selfCheckPending_.store(true);
            selfCheckRolledBack_.store(false);
            engineState_.store(static_cast<int32_t>(EngineState::SELF_CHECK_PENDING));
        } else {
            cfgSelfCheckDeadlineNs_ = 0;
            selfCheckPending_.store(false);
            selfCheckRolledBack_.store(false);
            engineState_.store(static_cast<int32_t>(EngineState::RUNNING_GRABBED));
        }
    }

    physCurrentSlot_ = 0;
    std::fill(std::begin(physToUinputSlot_), std::end(physToUinputSlot_), -1);
    for (auto& s : uinputSlots_) {
        s = UinputSlotState{};
    }
    uinputCurrentSlot_ = -1;
    btnTouchSent_ = false;
    typeACurrent_ = TypeAContact{};
    typeAFrameContacts_.clear();
    typeAActiveUinputSlots_.clear();
    virtualTargets_.clear();
    pendingWrites_.clear();

    threadRunning_.store(true);
    workerThread_ = std::thread(&TouchMuxEngine::engineLoop, this);
    signalWake();
    LOGI("TouchMux initialized & grabbed %s (%s), selfCheck=%dms",
         desc.path.c_str(), desc.name.c_str(), selfCheckTimeoutMs);
    return 0;
}

bool TouchMuxEngine::confirmSelfCheck() {
    std::lock_guard<std::mutex> lock(stateMutex_);
    if (!threadRunning_.load() || selfCheckRolledBack_.load()) {
        return false;
    }
    if (engineState_.load() == static_cast<int32_t>(EngineState::ERROR_STATE)) {
        return false;
    }
    cfgSelfCheckDeadlineNs_ = 0;
    cfgCancelSelfCheckRequested_ = true;
    selfCheckPending_.store(false);
    engineState_.store(static_cast<int32_t>(EngineState::RUNNING_GRABBED));
    signalWake();
    LOGI("TouchMux 5s self-check confirmed by user.");
    return true;
}

void TouchMuxEngine::shutdown() {
    if (threadRunning_.load()) {
        {
            std::lock_guard<std::mutex> lock(stateMutex_);
            cfgStopThreadRequested_ = true;
            cfgClickingRunning_ = false;
        }
        signalWake();
    }
    if (workerThread_.joinable()) {
        workerThread_.join();
    }
    selfCheckPending_.store(false);
    engineState_.store(static_cast<int32_t>(EngineState::IDLE));
}

void TouchMuxEngine::setDisplayMetrics(int displayW, int displayH, int rotation) {
    {
        std::lock_guard<std::mutex> lock(stateMutex_);
        cfgDisplayW_ = std::max(1, displayW);
        cfgDisplayH_ = std::max(1, displayH);
        cfgRotation_ = rotation;
    }
    signalWake();
}

void TouchMuxEngine::setTargetsAndRunning(const std::vector<NativeClickTarget>& targets,
                                          bool running,
                                          int displayW,
                                          int displayH,
                                          int rotation,
                                          float jitterRadiusPx) {

        std::lock_guard<std::mutex> lock(stateMutex_);
        cfgTargets_ = targets;
        cfgClickingRunning_ = running;
        cfgDisplayW_ = std::max(1, displayW);
        cfgDisplayH_ = std::max(1, displayH);
        cfgRotation_ = rotation;
        cfgJitterRadiusPx_ = std::clamp(jitterRadiusPx, 0.0f, 20.0f);
    }

}

void TouchMuxEngine::stopClicking() {
    {
        std::lock_guard<std::mutex> lock(stateMutex_);
        cfgClickingRunning_ = false;
    }
    signalWake();
}

int32_t TouchMuxEngine::nextTrackingId() {
    trackingCounter_++;
    if (trackingCounter_ > 65000) {
        trackingCounter_ = 100;
    }
    return trackingCounter_;
}

int TouchMuxEngine::countActiveUinputSlots() const {
    int cnt = 0;
    for (const auto& s : uinputSlots_) {
        if (s.owner != SlotOwner::FREE) cnt++;
    }
    return cnt;
}

int TouchMuxEngine::allocatePhysicalUinputSlot(int physSlot) {
    if (physSlot >= 0 && physSlot < kMaxUinputSlots && uinputSlots_[physSlot].owner == SlotOwner::FREE) {
        uinputSlots_[physSlot].owner = SlotOwner::PHYSICAL;
        uinputSlots_[physSlot].trackingId = nextTrackingId();
        uinputSlots_[physSlot].ownerSlotOrTargetId = physSlot;
        return physSlot;
    }
    for (int i = 0; i < kMaxUinputSlots; ++i) {
        if (uinputSlots_[i].owner == SlotOwner::FREE) {
            uinputSlots_[i].owner = SlotOwner::PHYSICAL;
            uinputSlots_[i].trackingId = nextTrackingId();
            uinputSlots_[i].ownerSlotOrTargetId = physSlot;
            return i;
        }
    }
    return -1;
}

int TouchMuxEngine::allocateVirtualUinputSlot(int32_t targetId) {
    // Allocate from high slots downwards (15..0) to avoid physical slot indices (0..9)
    for (int i = kMaxUinputSlots - 1; i >= 0; --i) {
        if (uinputSlots_[i].owner == SlotOwner::FREE) {
            uinputSlots_[i].owner = SlotOwner::VIRTUAL;
            uinputSlots_[i].trackingId = nextTrackingId();
            uinputSlots_[i].ownerSlotOrTargetId = targetId;
            return i;
        }
    }
    return -1;
}

void TouchMuxEngine::releaseUinputSlot(int uinputSlot) {
    if (uinputSlot >= 0 && uinputSlot < kMaxUinputSlots) {
        uinputSlots_[uinputSlot] = UinputSlotState{};
    }
}

void TouchMuxEngine::queueEvent(uint16_t type, uint16_t code, int32_t value) {
    struct input_event ev {};
    ev.type = type;
    ev.code = code;
    ev.value = value;
    pendingWrites_.push_back(ev);
}

void TouchMuxEngine::ensureUinputSlot(int uinputSlot) {
    if (uinputCurrentSlot_ != uinputSlot) {
        queueEvent(EV_ABS, ABS_MT_SLOT, uinputSlot);
        uinputCurrentSlot_ = uinputSlot;
    }
}

void TouchMuxEngine::emitSyncAndFlush() {
    if (pendingWrites_.empty()) {
        return;
    }
    const bool anyActive = countActiveUinputSlots() > 0;
    if (anyActive != btnTouchSent_) {
        queueEvent(EV_KEY, BTN_TOUCH, anyActive ? 1 : 0);
        if (deviceDesc_.hasBtnToolFinger) {
            queueEvent(EV_KEY, BTN_TOOL_FINGER, anyActive ? 1 : 0);
        }
        btnTouchSent_ = anyActive;
    }
    queueEvent(EV_SYN, SYN_REPORT, 0);

    const uint8_t* ptr = reinterpret_cast<const uint8_t*>(pendingWrites_.data());
    size_t remaining = pendingWrites_.size() * sizeof(struct input_event);
    while (remaining > 0) {
        ssize_t written = write(uinputFd_, ptr, remaining);
        if (written > 0) {
            ptr += written;
            remaining -= static_cast<size_t>(written);
        } else if (errno == EINTR) {
            continue;
        } else {
            break;
        }
    }
    pendingWrites_.clear();
}

void TouchMuxEngine::releaseAllVirtualSlotsAndSync() {
    bool changed = false;
    for (auto& vt : virtualTargets_) {
        if (vt.isPressed && vt.uinputSlot >= 0) {
            ensureUinputSlot(vt.uinputSlot);
            queueEvent(EV_ABS, ABS_MT_TRACKING_ID, -1);
            releaseUinputSlot(vt.uinputSlot);
            vt.uinputSlot = -1;
            vt.isPressed = false;
            changed = true;
        }
    }
    for (int i = 0; i < kMaxUinputSlots; ++i) {
        if (uinputSlots_[i].owner == SlotOwner::VIRTUAL) {
            ensureUinputSlot(i);
            queueEvent(EV_ABS, ABS_MT_TRACKING_ID, -1);
            releaseUinputSlot(i);
            changed = true;
        }
    }
    if (changed) {
        emitSyncAndFlush();
    }
}

void TouchMuxEngine::forceReleaseAllSlotsAndSync() {
    bool changed = false;
    for (int i = 0; i < kMaxUinputSlots; ++i) {
        if (uinputSlots_[i].owner != SlotOwner::FREE) {
            ensureUinputSlot(i);
            queueEvent(EV_ABS, ABS_MT_TRACKING_ID, -1);
            releaseUinputSlot(i);
            changed = true;
        }
    }
    std::fill(std::begin(physToUinputSlot_), std::end(physToUinputSlot_), -1);
    typeAActiveUinputSlots_.clear();
    for (auto& vt : virtualTargets_) {
        vt.isPressed = false;
        vt.uinputSlot = -1;
    }
    if (changed || btnTouchSent_) {
        emitSyncAndFlush();
    }
}

void TouchMuxEngine::logicalToRaw(float lx, float ly, int32_t& outRx, int32_t& outRy) {
    const float maxDispX = static_cast<float>(std::max(1, activeDisplayW_ - 1));
    const float maxDispY = static_cast<float>(std::max(1, activeDisplayH_ - 1));
    const float u = std::clamp(lx / maxDispX, 0.0f, 1.0f);
    const float v = std::clamp(ly / maxDispY, 0.0f, 1.0f);

    float uRaw = u;
    float vRaw = v;
    switch (activeRotation_) {
        case 1:  // Surface.ROTATION_90
            uRaw = 1.0f - v;
            vRaw = u;
            break;
        case 2:  // Surface.ROTATION_180
            uRaw = 1.0f - u;
            vRaw = 1.0f - v;
            break;
        case 3:  // Surface.ROTATION_270
            uRaw = v;
            vRaw = 1.0f - u;
            break;
        default: // Surface.ROTATION_0
            uRaw = u;
            vRaw = v;
            break;
    }

    const int64_t spanX = static_cast<int64_t>(deviceDesc_.absX.maximum) - deviceDesc_.absX.minimum;
    const int64_t spanY = static_cast<int64_t>(deviceDesc_.absY.maximum) - deviceDesc_.absY.minimum;
    int32_t rx = deviceDesc_.absX.minimum + static_cast<int32_t>(std::llround(uRaw * static_cast<double>(spanX)));
    int32_t ry = deviceDesc_.absY.minimum + static_cast<int32_t>(std::llround(vRaw * static_cast<double>(spanY)));

    outRx = std::clamp(rx, deviceDesc_.absX.minimum, deviceDesc_.absX.maximum);
    outRy = std::clamp(ry, deviceDesc_.absY.minimum, deviceDesc_.absY.maximum);
}

void TouchMuxEngine::handleWakeSignal() {
    uint64_t drained = 0;
    while (read(wakeFd_, &drained, sizeof(drained)) > 0) {
    }

    std::vector<NativeClickTarget> nextTargets;
    bool nextRunning = false;
    bool stopThread = false;
    {
        std::lock_guard<std::mutex> lock(stateMutex_);
        activeDisplayW_ = cfgDisplayW_;
        activeDisplayH_ = cfgDisplayH_;
        activeRotation_ = cfgRotation_;
        activeJitterRadiusPx_ = cfgJitterRadiusPx_;
        nextRunning = cfgClickingRunning_;
        nextTargets = cfgTargets_;
        stopThread = cfgStopThreadRequested_;
        if (cfgCancelSelfCheckRequested_) {
            selfCheckDeadlineNs_ = 0;
            cfgCancelSelfCheckRequested_ = false;
        } else if (selfCheckPending_.load()) {
            selfCheckDeadlineNs_ = cfgSelfCheckDeadlineNs_;
        }
    }

    if (stopThread) {
        threadRunning_.store(false);
        return;
    }

    if (!nextRunning) {
        activeClickingRunning_ = false;
        releaseAllVirtualSlotsAndSync();
        virtualTargets_.clear();
        for (const auto& t : nextTargets) {
            if (!t.enabled) continue;
            VirtualTargetRuntime rt;
            rt.config = t;
            virtualTargets_.push_back(rt);
        }
        rearmTimerFd();
        return;
    }

    const int64_t now = nowMonotonicNs();
    std::vector<VirtualTargetRuntime> rebuilt;
    rebuilt.reserve(nextTargets.size());

    // Keep state for existing targets; release virtual slots of removed/disabled targets
    for (auto& oldRt : virtualTargets_) {
        bool stillEnabled = false;
        for (const auto& nt : nextTargets) {
            if (nt.id == oldRt.config.id && nt.enabled) {
                stillEnabled = true;
                break;
            }
        }
        if (!stillEnabled && oldRt.isPressed && oldRt.uinputSlot >= 0) {
            ensureUinputSlot(oldRt.uinputSlot);
            queueEvent(EV_ABS, ABS_MT_TRACKING_ID, -1);
            releaseUinputSlot(oldRt.uinputSlot);
            oldRt.uinputSlot = -1;
            oldRt.isPressed = false;
        }
    }
    emitSyncAndFlush();

    for (size_t i = 0; i < nextTargets.size(); ++i) {
        const auto& nt = nextTargets[i];
        if (!nt.enabled) continue;
        bool matched = false;
        for (const auto& oldRt : virtualTargets_) {
            if (oldRt.config.id == nt.id && activeClickingRunning_) {
                VirtualTargetRuntime updated = oldRt;
                updated.config = nt;
                rebuilt.push_back(updated);
                matched = true;
                break;
            }
        }
        if (!matched) {
            VirtualTargetRuntime fresh;
            fresh.config = nt;
            fresh.isPressed = false;
            fresh.uinputSlot = -1;
            fresh.nextDeadlineNs = now + static_cast<int64_t>(i) * 500000LL;
            rebuilt.push_back(fresh);
        }
    }

    virtualTargets_ = std::move(rebuilt);
    activeClickingRunning_ = true;

    // Immediately fire targets due now
    handleTimerEvents();
}

void TouchMuxEngine::handleTypeAFrameEnd() {
    // Map Type-A contacts to Type-B uinput slots cleanly
    const size_t newCount = std::min(typeAFrameContacts_.size(), static_cast<size_t>(kMaxUinputSlots));
    while (typeAActiveUinputSlots_.size() > newCount) {
        int slot = typeAActiveUinputSlots_.back();
        typeAActiveUinputSlots_.pop_back();
        ensureUinputSlot(slot);
        queueEvent(EV_ABS, ABS_MT_TRACKING_ID, -1);
        releaseUinputSlot(slot);
    }
    while (typeAActiveUinputSlots_.size() < newCount) {
        int slot = allocatePhysicalUinputSlot(static_cast<int>(typeAActiveUinputSlots_.size()));
        if (slot < 0) break;
        typeAActiveUinputSlots_.push_back(slot);
        ensureUinputSlot(slot);
        queueEvent(EV_ABS, ABS_MT_TRACKING_ID, uinputSlots_[slot].trackingId);
    }
    for (size_t i = 0; i < typeAActiveUinputSlots_.size(); ++i) {
        int slot = typeAActiveUinputSlots_[i];
        const auto& c = typeAFrameContacts_[i];
        ensureUinputSlot(slot);
        if (c.hasX) queueEvent(EV_ABS, ABS_MT_POSITION_X, c.x);
        if (c.hasY) queueEvent(EV_ABS, ABS_MT_POSITION_Y, c.y);
        if (deviceDesc_.hasTouchMajor && c.major > 0) queueEvent(EV_ABS, ABS_MT_TOUCH_MAJOR, c.major);
        if (deviceDesc_.hasPressure && c.pressure > 0) queueEvent(EV_ABS, ABS_MT_PRESSURE, c.pressure);
    }
    typeAFrameContacts_.clear();
    typeACurrent_ = TypeAContact{};
    emitSyncAndFlush();
}

void TouchMuxEngine::handlePhysicalEvents() {
    struct input_event buf[64];
    while (true) {
        ssize_t n = read(physFd_, buf, sizeof(buf));
        if (n < 0) {
            if (errno == EAGAIN || errno == EWOULDBLOCK) {
                break;
            }
            if (errno == EINTR) {
                continue;
            }
            LOGE("read(physFd_) error: %s", std::strerror(errno));
            break;
        }
        if (n == 0) {
            break;
        }
        const size_t count = static_cast<size_t>(n) / sizeof(struct input_event);
        for (size_t i = 0; i < count; ++i) {
            const struct input_event& ev = buf[i];

            if (!deviceDesc_.isTypeB) {
                if (ev.type == EV_ABS) {
                    if (ev.code == ABS_MT_POSITION_X) {
                        typeACurrent_.hasX = true;
                        typeACurrent_.x = ev.value;
                    } else if (ev.code == ABS_MT_POSITION_Y) {
                        typeACurrent_.hasY = true;
                        typeACurrent_.y = ev.value;
                    } else if (ev.code == ABS_MT_TOUCH_MAJOR) {
                        typeACurrent_.major = ev.value;
                    } else if (ev.code == ABS_MT_PRESSURE) {
                        typeACurrent_.pressure = ev.value;
                    }
                } else if (ev.type == EV_SYN && ev.code == SYN_MT_REPORT) {
                    if (typeACurrent_.hasX || typeACurrent_.hasY) {
                        typeAFrameContacts_.push_back(typeACurrent_);
                    }
                    typeACurrent_ = TypeAContact{};
                } else if (ev.type == EV_SYN && ev.code == SYN_REPORT) {
                    handleTypeAFrameEnd();
                } else if (ev.type == EV_SYN && ev.code == SYN_DROPPED) {
                    typeAFrameContacts_.clear();
                    typeACurrent_ = TypeAContact{};
                    handleTypeAFrameEnd();
                }
                continue;
            }

            // Type-B physical touchscreen handling
            if (ev.type == EV_SYN) {
                if (ev.code == SYN_REPORT) {
                    emitSyncAndFlush();
                } else if (ev.code == SYN_DROPPED) {
                    LOGW("SYN_DROPPED on physical touchscreen; resetting physical slots.");
                    for (int p = 0; p < kMaxPhysSlots; ++p) {
                        int u = physToUinputSlot_[p];
                        if (u >= 0) {
                            ensureUinputSlot(u);
                            queueEvent(EV_ABS, ABS_MT_TRACKING_ID, -1);
                            releaseUinputSlot(u);
                            physToUinputSlot_[p] = -1;
                        }
                    }
                    emitSyncAndFlush();
                }
                continue;
            }

            if (ev.type == EV_KEY) {
                if (ev.code == BTN_TOUCH || ev.code == BTN_TOOL_FINGER) {
                    // Computed dynamically in emitSyncAndFlush() from total active slots
                    continue;
                }
                continue;
            }

            if (ev.type == EV_ABS) {
                if (ev.code == ABS_MT_SLOT) {
                    physCurrentSlot_ = std::clamp(ev.value, 0, kMaxPhysSlots - 1);
                } else if (ev.code == ABS_MT_TRACKING_ID) {
                    if (ev.value >= 0) {
                        int u = physToUinputSlot_[physCurrentSlot_];
                        if (u < 0) {
                            u = allocatePhysicalUinputSlot(physCurrentSlot_);
                            physToUinputSlot_[physCurrentSlot_] = u;
                        } else {
                            // Reassign tracking ID if kernel replaced tracking ID directly
                            uinputSlots_[u].trackingId = nextTrackingId();
                        }
                        if (u >= 0) {
                            ensureUinputSlot(u);
                            queueEvent(EV_ABS, ABS_MT_TRACKING_ID, uinputSlots_[u].trackingId);
                        }
                    } else {
                        int u = physToUinputSlot_[physCurrentSlot_];
                        if (u >= 0) {
                            ensureUinputSlot(u);
                            queueEvent(EV_ABS, ABS_MT_TRACKING_ID, -1);
                            releaseUinputSlot(u);
                            physToUinputSlot_[physCurrentSlot_] = -1;
                        }
                    }
                } else {
                    // Coordinates or touch metrics for current physical slot
                    int u = physToUinputSlot_[physCurrentSlot_];
                    if (u < 0 && (ev.code == ABS_MT_POSITION_X || ev.code == ABS_MT_POSITION_Y)) {
                        // Finger was already down prior to EVIOCGRAB; adopt smoothly
                        u = allocatePhysicalUinputSlot(physCurrentSlot_);
                        physToUinputSlot_[physCurrentSlot_] = u;
                        if (u >= 0) {
                            ensureUinputSlot(u);
                            queueEvent(EV_ABS, ABS_MT_TRACKING_ID, uinputSlots_[u].trackingId);
                        }
                    }
                    if (u >= 0) {
                        ensureUinputSlot(u);
                        queueEvent(EV_ABS, ev.code, ev.value);
                    }
                }
            }
        }
    }
}

void TouchMuxEngine::handleTimerEvents() {
    uint64_t expirations = 0;
    while (read(timerFd_, &expirations, sizeof(expirations)) > 0) {
    }

    const int64_t now = nowMonotonicNs();

    if (selfCheckPending_.load() && selfCheckDeadlineNs_ > 0 && now >= selfCheckDeadlineNs_) {
        LOGW("TouchMux 5s self-check timed out! Rolling back EVIOCGRAB safely.");
        selfCheckPending_.store(false);
        selfCheckRolledBack_.store(true);
        activeClickingRunning_ = false;
        threadRunning_.store(false);
        engineState_.store(static_cast<int32_t>(EngineState::IDLE));
        return;
    }

    if (activeClickingRunning_) {
        std::uniform_real_distribution<float> jitterDist(-activeJitterRadiusPx_, activeJitterRadiusPx_);
        bool anyEmitted = false;

        for (auto& vt : virtualTargets_) {
            if (!vt.config.enabled) continue;
            if (vt.nextDeadlineNs > now) continue;

            if (!vt.isPressed) {
                int slot = allocateVirtualUinputSlot(vt.config.id);
                if (slot < 0) {
                    // All 16 slots currently occupied; retry shortly without disturbing physical touches
                    vt.nextDeadlineNs = now + 5000000LL;
                    continue;
                }
                float targetX = vt.config.x;
                float targetY = vt.config.y;
                if (vt.config.microJitter && activeJitterRadiusPx_ > 0.0f) {
                    targetX += jitterDist(rng_);
                    targetY += jitterDist(rng_);
                }
                int32_t rx = 0;
                int32_t ry = 0;
                logicalToRaw(targetX, targetY, rx, ry);

                ensureUinputSlot(slot);
                queueEvent(EV_ABS, ABS_MT_TRACKING_ID, uinputSlots_[slot].trackingId);
                queueEvent(EV_ABS, ABS_MT_POSITION_X, rx);
                queueEvent(EV_ABS, ABS_MT_POSITION_Y, ry);
                if (deviceDesc_.hasTouchMajor) {
                    int32_t major = std::clamp(8, deviceDesc_.absTouchMajor.minimum,
                                               std::max(deviceDesc_.absTouchMajor.minimum, deviceDesc_.absTouchMajor.maximum));
                    queueEvent(EV_ABS, ABS_MT_TOUCH_MAJOR, major);
                }
                if (deviceDesc_.hasPressure) {
                    int32_t pressure = std::clamp(64, deviceDesc_.absPressure.minimum,
                                                  std::max(deviceDesc_.absPressure.minimum, deviceDesc_.absPressure.maximum));
                    queueEvent(EV_ABS, ABS_MT_PRESSURE, pressure);
                }

                vt.uinputSlot = slot;
                vt.isPressed = true;
                int64_t pressNs = std::clamp(vt.config.pressDurationMs, int64_t{8}, int64_t{5000}) * 1000000LL;
                vt.nextDeadlineNs = now + pressNs;
                anyEmitted = true;
            } else {
                if (vt.uinputSlot >= 0) {
                    ensureUinputSlot(vt.uinputSlot);
                    queueEvent(EV_ABS, ABS_MT_TRACKING_ID, -1);
                    releaseUinputSlot(vt.uinputSlot);
                }
                vt.uinputSlot = -1;
                vt.isPressed = false;
                int64_t intervalNs = std::clamp(vt.config.intervalMs, int64_t{5}, int64_t{60000}) * 1000000LL;
                vt.nextDeadlineNs = now + intervalNs;
                anyEmitted = true;
            }
        }

        if (anyEmitted) {
            emitSyncAndFlush();
        }
    }

    rearmTimerFd();
}

void TouchMuxEngine::rearmTimerFd() {
    if (timerFd_ < 0) return;

    int64_t earliestNs = 0;
    if (selfCheckPending_.load() && selfCheckDeadlineNs_ > 0) {
        earliestNs = selfCheckDeadlineNs_;
    }
    if (activeClickingRunning_) {
        for (const auto& vt : virtualTargets_) {
            if (!vt.config.enabled) continue;
            if (earliestNs == 0 || vt.nextDeadlineNs < earliestNs) {
                earliestNs = vt.nextDeadlineNs;
            }
        }
    }

    struct itimerspec its {};
    if (earliestNs > 0) {
        const int64_t now = nowMonotonicNs();
        int64_t deltaNs = earliestNs - now;
        if (deltaNs < 100000LL) {
            deltaNs = 100000LL;  // Minimum 100us so timerfd is always armed in the future
        }
        its.it_value.tv_sec = static_cast<time_t>(deltaNs / 1000000000LL);
        its.it_value.tv_nsec = static_cast<long>(deltaNs % 1000000000LL);
    }
    timerfd_settime(timerFd_, 0, &its, nullptr);
}

void TouchMuxEngine::engineLoop() {
    setpriority(PRIO_PROCESS, 0, -10);

    constexpr int kMaxEvents = 8;
    struct epoll_event events[kMaxEvents];

    while (threadRunning_.load()) {
        int nfds = epoll_wait(epollFd_, events, kMaxEvents, -1);
        if (nfds < 0) {
            if (errno == EINTR) continue;
            LOGE("epoll_wait failed: %s", std::strerror(errno));
            break;
        }
        for (int i = 0; i < nfds; ++i) {
            int fd = events[i].data.fd;
            if (fd == physFd_) {
                handlePhysicalEvents();
            } else if (fd == timerFd_) {
                handleTimerEvents();
            } else if (fd == wakeFd_) {
                handleWakeSignal();
            }
        }
    }

    forceReleaseAllSlotsAndSync();
    if (physFd_ >= 0) {
        ioctl(physFd_, EVIOCGRAB, 0);
    }
    if (uinputFd_ >= 0) {
        ioctl(uinputFd_, UI_DEV_DESTROY);
    }
    closeAllFds();
    LOGI("TouchMux engineLoop exited cleanly and released EVIOCGRAB.");
}

    fun deletePreset(context: Context, presetId: String): Boolean {
        val presets = loadPresets(context)
        if (presets.size <= 1) return false
        val idx = presets.indexOfFirst { it.id == presetId }
        if (idx < 0) return false
        val currentActive = getActivePresetId(context)
        presets.removeAt(idx)
        val nextActive = if (currentActive == presetId) {
            presets[idx.coerceAtMost(presets.lastIndex)].id
        } else {
            currentActive
        }
        persistPresetsInternal(context, presets, nextActive)
        return true
    }

    fun switchToNextPreset(context: Context): Triple<Int, Int, ClickPreset> {
        val presets = loadPresets(context)
        val activeId = getActivePresetId(context)
        val currentIdx = presets.indexOfFirst { it.id == activeId }.coerceAtLeast(0)
        val nextIdx = (currentIdx + 1) % presets.size
        val nextPreset = presets[nextIdx]
        persistPresetsInternal(context, presets, nextPreset.id)
        return Triple(nextIdx, presets.size, nextPreset)
    }

    fun loadTargets(context: Context): MutableList<ClickTarget> =
        getActivePreset(context).targets.map { it.copy() }.toMutableList()

    fun saveTargets(context: Context, targets: List<ClickTarget>) {
        val presets = loadPresets(context)
        val activeId = getActivePresetId(context)
        val active = presets.firstOrNull { it.id == activeId } ?: presets.first()
        active.targets.clear()
        targets.forEachIndexed { idx, t ->
            t.index = idx + 1
            active.targets += t.copy(index = idx + 1)
        }
        persistPresetsInternal(context, presets, active.id)
    }

        if (!prefs.contains(KEY_CONTROL_BAR_X) || !prefs.contains(KEY_CONTROL_BAR_Y)) {

        }

    }

    override fun onInterceptTouchEvent(ev: MotionEvent): Boolean {
        when (ev.actionMasked) {
            MotionEvent.ACTION_DOWN -> {
                downRawX = ev.rawX
                downRawY = ev.rawY
                startLpX = currentLpX
                startLpY = currentLpY
                isDragging = false
                return false
            }
            MotionEvent.ACTION_MOVE -> {
                val dx = ev.rawX - downRawX
                val dy = ev.rawY - downRawY
                if (hypot(dx, dy) >= touchSlop) {
                    isDragging = true
                    listener.onDragStart()
                    return true
                }
            }
        }
        return false
    }

    @SuppressLint("ClickableViewAccessibility")
    override fun onTouchEvent(event: MotionEvent): Boolean {
        when (event.actionMasked) {
            MotionEvent.ACTION_DOWN -> {
                downRawX = event.rawX
                downRawY = event.rawY
                startLpX = currentLpX
                startLpY = currentLpY
                isDragging = false
                return true
            }
            MotionEvent.ACTION_MOVE -> {
                val dx = event.rawX - downRawX
                val dy = event.rawY - downRawY
                if (!isDragging && hypot(dx, dy) >= touchSlop) {
                    isDragging = true
                    listener.onDragStart()
                }
                if (isDragging) {
                    val nx = (startLpX + dx).roundToInt()
                    val ny = (startLpY + dy).roundToInt()
                    currentLpX = nx
                    currentLpY = ny
                    listener.onDragMove(nx, ny)
                }
                return true
            }
            MotionEvent.ACTION_UP, MotionEvent.ACTION_CANCEL -> {
                if (isDragging) {
                    listener.onDragEnd()
                }
                isDragging = false
                return true
            }
        }
        return super.onTouchEvent(event)
    }

        override fun onDraw(canvas: Canvas) {
            super.onDraw(canvas)
            val w = width.toFloat()
            val h = height.toFloat()
            if (w <= 0f || h <= 0f) return

            val inset = 0.8f * density
            val r = 16f * density
            val radii = if (dockRightState) {
                floatArrayOf(r, r, 0f, 0f, 0f, 0f, r, r)
            } else {
                floatArrayOf(0f, 0f, r, r, r, r, 0f, 0f)
            }

            path.reset()
            rect.set(
                if (dockRightState) inset else -inset,
                inset,
                if (dockRightState) w + inset else w - inset,
                h - inset
            )
            path.addRoundRect(rect, radii, Path.Direction.CW)

            fillPaint.color = colors.surfaceContainerHigh
            canvas.drawPath(path, fillPaint)

            borderPaint.color = colors.outlineVariant
            canvas.drawPath(path, borderPaint)

            val cx = w * 0.5f
            val ledY = h * 0.34f
            val ledColor = statusLedColor()
            if (runningState) {
                haloPaint.color = ColorUtils.setAlphaComponent(ledColor, 95)
                canvas.drawCircle(cx, ledY, 7.2f * density, haloPaint)
            }
            ledPaint.color = ledColor
            canvas.drawCircle(cx, ledY, 4.0f * density, ledPaint)

            // Expand chevron pointing inward from the docked edge
            chevronPaint.color = colors.onSurface
            val chevY = h * 0.66f
            val aw = 3.5f * density
            val ah = 4.6f * density
            if (dockRightState) {
                canvas.drawLine(cx + aw * 0.5f, chevY - ah, cx - aw * 0.6f, chevY, chevronPaint)
                canvas.drawLine(cx + aw * 0.5f, chevY + ah, cx - aw * 0.6f, chevY, chevronPaint)
            } else {
                canvas.drawLine(cx - aw * 0.5f, chevY - ah, cx + aw * 0.6f, chevY, chevronPaint)
                canvas.drawLine(cx - aw * 0.5f, chevY + ah, cx + aw * 0.6f, chevY, chevronPaint)
            }
        }
    }
}

/**
 * Foreground overlay service implementing zero-relayout window orchestration.
 *
 * Key properties:
 * 1. All overlay windows use FLAG_NOT_FOCUSABLE | FLAG_SPLIT_TOUCH | FLAG_LAYOUT_IN_SCREEN | FLAG_LAYOUT_NO_LIMITS.
 * 2. When locked, anchor windows carry FLAG_NOT_TOUCHABLE. Toggling Start/Stop NEVER calls
 *    WindowManager.updateViewLayout() or triggers WMS relayout — only local View.invalidate().
 * 3. Tracks DisplayManager rotation changes in real time for landscape games.
 */
class OverlayService : Service(), ControlBarView.Listener {

    private lateinit var windowManager: WindowManager
    private lateinit var displayManager: DisplayManager
    private lateinit var themeColors: OverlayThemeColors
    private val serviceScope = CoroutineScope(SupervisorJob() + Dispatchers.Main.immediate)
    private val mainHandler = Handler(Looper.getMainLooper())
    private val autoCollapseRunnable = Runnable { collapseControlBar() }

    private var controlBarView: ControlBarView? = null
    private var controlBarLp: WindowManager.LayoutParams? = null
    private var isBarCollapsed: Boolean = false
    private var dockRight: Boolean = false

    private var tunePanelView: View? = null
    private var tunePanelLp: WindowManager.LayoutParams? = null

    private val targets = mutableListOf<ClickTarget>()
    private val anchorViews = LinkedHashMap<Int, AnchorPointView>()
    private val anchorLps = LinkedHashMap<Int, WindowManager.LayoutParams>()

    private var selectedTargetId: Int = -1
    private var nextTargetId: Int = 1
    private var activePresetId: String = ""
    private var isLocked: Boolean = false
    private var isClicking: Boolean = false
    private var engineMode: Int = TouchMuxUserService.MODE_IDLE
    private var engineCollectJob: Job? = null

    private val displayListener = object : DisplayManager.DisplayListener {
        override fun onDisplayAdded(displayId: Int) = Unit
        override fun onDisplayRemoved(displayId: Int) = Unit
        override fun onDisplayChanged(displayId: Int) {

        }
    }

    override fun onCreate() {
        super.onCreate()
        windowManager = getSystemService(Context.WINDOW_SERVICE) as WindowManager
        displayManager = getSystemService(Context.DISPLAY_SERVICE) as DisplayManager
        themeColors = OverlayThemeColors.resolve(this)
        dockRight = ClickConfigStore.loadControlBarDockRight(this)

            }
        }
    }

    override fun onStartCommand(intent: Intent?, flags: Int, startId: Int): Int {
        when (intent?.action) {
            ACTION_STOP_OVERLAY -> {
                stopClickingInternal()
                stopSelf()
                return START_NOT_STICKY
            }
            ACTION_STOP_CLICKING -> {
                stopClickingInternal()
            }
            ACTION_REFRESH_SETTINGS -> {
                reloadTargetSettingsFromStore()
                refreshControlBarState()
                if (ClickConfigStore.loadSettings(this).autoCollapseBar) {
                    scheduleAutoCollapse(IDLE_COLLAPSE_DELAY_MS)
                } else {
                    cancelAutoCollapse()
                }
            }
        }
        return START_STICKY
    }

    override fun onConfigurationChanged(newConfig: Configuration) {
        super.onConfigurationChanged(newConfig)
        themeColors = OverlayThemeColors.resolve(this)
        controlBarView?.applyThemeColors(themeColors)

        if (tunePanelView != null) {

        }
    }

    private fun startForegroundWithNotification() {

        }

        val openIntent = PendingIntent.getActivity(
            this,
            0,
            Intent(this, MainActivity::class.java).apply {
                flags = Intent.FLAG_ACTIVITY_NEW_TASK or Intent.FLAG_ACTIVITY_SINGLE_TOP
            },
            PendingIntent.FLAG_IMMUTABLE or PendingIntent.FLAG_UPDATE_CURRENT
        )

        val stopClickIntent = PendingIntent.getService(
            this,
            1,
            Intent(this, OverlayService::class.java).apply { action = ACTION_STOP_CLICKING },

            .setSmallIcon(R.drawable.ic_touch_app)

            .setContentIntent(openIntent)

    private fun createBaseOverlayParams(
        width: Int,
        height: Int,
        touchable: Boolean
    ): WindowManager.LayoutParams {
        var flags = WindowManager.LayoutParams.FLAG_NOT_FOCUSABLE or
            WindowManager.LayoutParams.FLAG_SPLIT_TOUCH or
            WindowManager.LayoutParams.FLAG_LAYOUT_IN_SCREEN or
            WindowManager.LayoutParams.FLAG_LAYOUT_NO_LIMITS or
            WindowManager.LayoutParams.FLAG_HARDWARE_ACCELERATED
        if (!touchable) {
            flags = flags or WindowManager.LayoutParams.FLAG_NOT_TOUCHABLE
        }
        return WindowManager.LayoutParams(
            width,
            height,
            WindowManager.LayoutParams.TYPE_APPLICATION_OVERLAY,
            flags,
            PixelFormat.TRANSLUCENT
        ).apply {
            gravity = Gravity.TOP or Gravity.START
            if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.R) {
                fitInsetsTypes = 0
                layoutInDisplayCutoutMode =
                    WindowManager.LayoutParams.LAYOUT_IN_DISPLAY_CUTOUT_MODE_ALWAYS
            } else if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.P) {
                layoutInDisplayCutoutMode =
                    WindowManager.LayoutParams.LAYOUT_IN_DISPLAY_CUTOUT_MODE_SHORT_EDGES
            }
        }
    }

    private fun loadInitialTargets() {
        activePresetId = ClickConfigStore.getActivePresetId(this)
        val loaded = ClickConfigStore.loadTargets(this)
        targets.clear()
        targets.addAll(loaded)
        if (targets.isEmpty()) {
            val metrics = ShizukuController.getDisplayScreenMetrics(this)
            val settings = ClickConfigStore.loadSettings(this)
            targets += ClickTarget(
                id = 1,
                index = 1,
                x = metrics.widthPx * 0.5f,
                y = metrics.heightPx * 0.5f,
                intervalMs = settings.defaultIntervalMs,
                pressDurationMs = settings.defaultPressDurationMs,
                microJitter = settings.defaultMicroJitter,
                enabled = true
            )
            persistCurrentTargets()
        }
        nextTargetId = (targets.maxOfOrNull { it.id } ?: 0) + 1
        selectedTargetId = targets.firstOrNull()?.id ?: -1
    }

    private fun attachControlBar() {
        val bar = ControlBarView(this, themeColors, this)
        val metrics = ShizukuController.getDisplayScreenMetrics(this)
        val savedPos = ClickConfigStore.loadControlBarPosition(this)
        bar.setToolsExpanded(false)
        val initialW = bar.getExpandedWidthPx()
        val initialH = bar.getExpandedBarHeightPx()
        val defaultX = if (dockRight) {
            (metrics.widthPx - initialW - (16f * resources.displayMetrics.density).roundToInt())
                .coerceAtLeast(0)
        } else {
            (16f * resources.displayMetrics.density).roundToInt()
        }
        val lp = createBaseOverlayParams(
            initialW,
            initialH,
            touchable = true
        ).apply {
            x = (savedPos?.first ?: defaultX)
                .coerceIn(0, (metrics.widthPx - initialW).coerceAtLeast(0))
            y = (savedPos?.second ?: (metrics.heightPx * 0.25f).roundToInt())
                .coerceIn(0, (metrics.heightPx - initialH).coerceAtLeast(0))
        }
        dockRight = (lp.x + initialW / 2) >= (metrics.widthPx / 2)
        bar.setCollapsed(false, dockRight)
        bar.syncWindowPosition(lp.x, lp.y)
        windowManager.addView(bar, lp)
        controlBarView = bar
        controlBarLp = lp
        isBarCollapsed = false
        refreshControlBarState()
    }

    private fun reloadTargetSettingsFromStore() {
        val storeActiveId = ClickConfigStore.getActivePresetId(this)
        val stored = ClickConfigStore.loadTargets(this)
        val currentSig = targets.map { listOf(it.id, it.x, it.y, it.intervalMs, it.pressDurationMs, it.microJitter, it.enabled) }
        val storedSig = stored.map { listOf(it.id, it.x, it.y, it.intervalMs, it.pressDurationMs, it.microJitter, it.enabled) }
        if (activePresetId != storeActiveId || currentSig != storedSig) {
            if (isClicking) {
                stopClickingInternal()
            }
            replaceTargetsWithActivePreset()
        }
    }

    private fun replaceTargetsWithActivePreset() {
        activePresetId = ClickConfigStore.getActivePresetId(this)
        dismissTunePanel()
        removeAllAnchorWindows()
        targets.clear()
        targets.addAll(ClickConfigStore.loadTargets(this))
        nextTargetId = (targets.maxOfOrNull { it.id } ?: 0) + 1
        selectedTargetId = targets.firstOrNull()?.id ?: -1
        attachAllAnchors()
        refreshControlBarState()
    }

    private fun persistCurrentTargets() {
        ClickConfigStore.saveTargets(this, targets)
        OverlayStateHub.bumpConfigVersion()
    }

    private fun removeControlBar() {
        val bar = controlBarView ?: return
        try {
            windowManager.removeViewImmediate(bar)
        } catch (_: Throwable) {
        }
        controlBarView = null
        controlBarLp = null
    }

    private fun attachAllAnchors() {
        for (t in targets) {
            attachSingleAnchor(t)
        }
        refreshAnchorVisuals()
    }

    private fun attachSingleAnchor(target: ClickTarget) {
        if (anchorViews.containsKey(target.id)) return
        val metrics = ShizukuController.getDisplayScreenMetrics(this)
        val view = AnchorPointView(
            context = this,
            targetId = target.id,
            onDragMove = { id, nx, ny -> handleAnchorDragMove(id, nx, ny) },
            onDragFinish = { id -> handleAnchorDragFinish(id) },
            onTap = { id -> handleAnchorTap(id) }
        )
        target.x = target.x.coerceIn(0f, (metrics.widthPx - 1).coerceAtLeast(1).toFloat())
        target.y = target.y.coerceIn(0f, (metrics.heightPx - 1).coerceAtLeast(1).toFloat())

        val lp = createBaseOverlayParams(
            view.viewSizePx,
            view.viewSizePx,
            touchable = !isLocked
        ).apply {
            x = (target.x - view.halfSizePx).roundToInt()
            y = (target.y - view.halfSizePx).roundToInt()
            alpha = if (isLocked) 0.80f else 1.0f
        }
        view.syncWindowPosition(lp.x, lp.y)
        windowManager.addView(view, lp)
        anchorViews[target.id] = view
        anchorLps[target.id] = lp
    }

    private fun removeAllAnchorWindows() {
        for (v in anchorViews.values) {
            try {
                windowManager.removeViewImmediate(v)
            } catch (_: Throwable) {
            }
        }
        anchorViews.clear()
        anchorLps.clear()
    }

    private fun handleAnchorDragMove(targetId: Int, newLpX: Int, newLpY: Int) {
        val view = anchorViews[targetId] ?: return
        val lp = anchorLps[targetId] ?: return
        val metrics = ShizukuController.getDisplayScreenMetrics(this)
        val minX = -view.halfSizePx
        val maxX = metrics.widthPx - 1 - view.halfSizePx
        val minY = -view.halfSizePx
        val maxY = metrics.heightPx - 1 - view.halfSizePx

        lp.x = newLpX.coerceIn(minX, maxX.coerceAtLeast(minX))
        lp.y = newLpY.coerceIn(minY, maxY.coerceAtLeast(minY))
        windowManager.updateViewLayout(view, lp)

        targets.firstOrNull { it.id == targetId }?.let { t ->
            t.x = (lp.x + view.halfSizePx).toFloat()
            t.y = (lp.y + view.halfSizePx).toFloat()
        }
    }

    private fun handleAnchorDragFinish(targetId: Int) {
        selectedTargetId = targetId
        persistCurrentTargets()
        refreshAnchorVisuals()
        if (isClicking) {
            pushTargetsToServiceAsync()
        }
    }

    private fun handleAnchorTap(targetId: Int) {
        selectedTargetId = targetId
        refreshAnchorVisuals()
        showOrUpdateTunePanel(forceShow = true)
    }

    private fun refreshAnchorVisuals() {
        targets.forEachIndexed { idx, t ->
            t.index = idx + 1
            anchorViews[t.id]?.setVisualState(
                index = t.index,
                enabled = t.enabled,
                jitter = t.microJitter,
                isLocked = isLocked,
                isRunning = isClicking,
                isSelected = (t.id == selectedTargetId)
            )
        }
    }

    private fun refreshControlBarState() {
        val (presetIdx, presetCount) = ClickConfigStore.getActivePresetIndexAndTotal(this)
        controlBarView?.updateState(
            isRunning = isClicking,
            isLocked = isLocked,
            engineMode = engineMode,
            tunePanelOpen = (tunePanelView != null),
            presetIndex = presetIdx,
            presetCount = presetCount
        )
    }

    /**
     * Switches anchor windows between Edit (touchable) and Locked (FLAG_NOT_TOUCHABLE).
     * Only calls updateViewLayout when lock state actually changes.
     */
    private fun applyLockMode(locked: Boolean) {
        if (isLocked == locked) return
        isLocked = locked
        if (locked) {
            dismissTunePanel()
        }
        for ((id, view) in anchorViews) {
            val lp = anchorLps[id] ?: continue
            val updatedFlags = if (locked) {
                lp.flags or WindowManager.LayoutParams.FLAG_NOT_TOUCHABLE
            } else {
                lp.flags and WindowManager.LayoutParams.FLAG_NOT_TOUCHABLE.inv()
            }
            val updatedAlpha = if (locked) 0.80f else 1.0f
            if (lp.flags != updatedFlags || lp.alpha != updatedAlpha) {
                lp.flags = updatedFlags
                lp.alpha = updatedAlpha
                windowManager.updateViewLayout(view, lp)
            }
        }
        refreshAnchorVisuals()
        refreshControlBarState()
    }

    private fun scheduleAutoCollapse(delayMs: Long) {
        cancelAutoCollapse()
        if (isBarCollapsed) return
        if (tunePanelView != null) return
        if (!ClickConfigStore.loadSettings(this).autoCollapseBar) return
        mainHandler.postDelayed(autoCollapseRunnable, delayMs)
    }

    private fun cancelAutoCollapse() {
        mainHandler.removeCallbacks(autoCollapseRunnable)
    }

    private fun collapseControlBar() {
        if (isBarCollapsed) return
        val bar = controlBarView ?: return
        val lp = controlBarLp ?: return
        cancelAutoCollapse()
        dismissTunePanel()

        val metrics = ShizukuController.getDisplayScreenMetrics(this)
        dockRight = (lp.x + bar.getCurrentWidthPx() / 2) >= (metrics.widthPx / 2)
        isBarCollapsed = true
        bar.setCollapsed(true, dockRight)

        lp.width = bar.getCollapsedWidthPx()
        lp.height = bar.getCollapsedHeightPx()
        lp.x = if (dockRight) (metrics.widthPx - lp.width).coerceAtLeast(0) else 0
        lp.y = lp.y.coerceIn(0, (metrics.heightPx - lp.height).coerceAtLeast(0))
        bar.syncWindowPosition(lp.x, lp.y)
        windowManager.updateViewLayout(bar, lp)
        ClickConfigStore.saveControlBarCollapseState(this, true, dockRight)
    }

    private fun expandControlBar() {
        cancelAutoCollapse()
        val bar = controlBarView ?: return
        val lp = controlBarLp ?: return
        val metrics = ShizukuController.getDisplayScreenMetrics(this)
        val edgeMarginPx = (12f * resources.displayMetrics.density).roundToInt()

        isBarCollapsed = false
        bar.setCollapsed(false, dockRight)

        lp.width = bar.getCurrentWidthPx()
        lp.height = bar.getCurrentHeightPx()
        lp.x = if (dockRight) {
            (metrics.widthPx - lp.width - edgeMarginPx).coerceAtLeast(0)
        } else {
            edgeMarginPx.coerceAtMost((metrics.widthPx - lp.width).coerceAtLeast(0))
        }
        lp.y = lp.y.coerceIn(0, (metrics.heightPx - lp.height).coerceAtLeast(0))
        bar.syncWindowPosition(lp.x, lp.y)
        windowManager.updateViewLayout(bar, lp)
        ClickConfigStore.saveControlBarCollapseState(this, false, dockRight)
        ClickConfigStore.saveControlBarPosition(this, lp.x, lp.y)
        scheduleAutoCollapse(IDLE_COLLAPSE_DELAY_MS)
    }

    override fun onSwitchPreset() {
        val presets = ClickConfigStore.loadPresets(this)
        if (presets.size <= 1) {
            val only = presets.firstOrNull()
            Toast.makeText(
                this,
                "当前仅 1 个方案「${only?.name ?: "方案 1"}」，可在主界面新增方案",
                Toast.LENGTH_SHORT
            ).show()
            scheduleAutoCollapse(IDLE_COLLAPSE_DELAY_MS)
            return
        }
        if (isClicking) {
            stopClickingInternal()
        }
        val (newIdx, total, nextPreset) = ClickConfigStore.switchToNextPreset(this)
        replaceTargetsWithActivePreset()
        OverlayStateHub.bumpConfigVersion()
        Toast.makeText(
            this,
            "切换方案 ${newIdx + 1}/$total：${nextPreset.name} (${targets.size} 点)",
            Toast.LENGTH_SHORT
        ).show()
        scheduleAutoCollapse(IDLE_COLLAPSE_DELAY_MS)
    }

    override fun onToggleToolsExpanded() {
        cancelAutoCollapse()
        val bar = controlBarView ?: return
        val lp = controlBarLp ?: return
        val metrics = ShizukuController.getDisplayScreenMetrics(this)
        val nextExpanded = !bar.isToolsExpanded()
        bar.setToolsExpanded(nextExpanded)
        if (!nextExpanded) {
            dismissTunePanel()
        }
        lp.width = bar.getCurrentWidthPx()
        lp.height = bar.getCurrentHeightPx()
        lp.y = lp.y.coerceIn(0, (metrics.heightPx - lp.height).coerceAtLeast(0))
        bar.syncWindowPosition(lp.x, lp.y)
        windowManager.updateViewLayout(bar, lp)
        scheduleAutoCollapse(IDLE_COLLAPSE_DELAY_MS)
    }

    override fun onCollapseRequested() {
        collapseControlBar()
    }

    override fun onExpandRequested() {
        expandControlBar()
    }

    override fun onDragStart() {
        cancelAutoCollapse()
    }

    override fun onTogglePlayStop() {
        if (isClicking) {
            stopClickingInternal()
            return
        }
        if (targets.none { it.enabled }) {
            Toast.makeText(this, "请至少启用一个点击锚点", Toast.LENGTH_SHORT).show()
            scheduleAutoCollapse(IDLE_COLLAPSE_DELAY_MS)
            return
        }
        if (!ShizukuController.isServiceConnected()) {
            ShizukuController.evaluateShizukuState()
            Toast.makeText(this, "Shizuku 触控服务尚未连接，请在主界面授权并连接", Toast.LENGTH_SHORT).show()
            scheduleAutoCollapse(IDLE_COLLAPSE_DELAY_MS)
            return
        }

        // If already locked, zero updateViewLayout calls occur on Start!
        if (!isLocked) {
            applyLockMode(true)
            // Wait one animation frame after applying FLAG_NOT_TOUCHABLE before injecting first tap
            controlBarView?.postOnAnimation {
                startClickingInternal()
            } ?: startClickingInternal()
        } else {
            startClickingInternal()
        }
    }

    private fun startClickingInternal() {
        cancelAutoCollapse()
        val snapshot = targets.map { it.copy() }
        isClicking = true
        OverlayStateHub.setClickingRunning(true)
        refreshAnchorVisuals()
        refreshControlBarState()

        serviceScope.launch(Dispatchers.IO) {
            val mode = ShizukuController.startClicking(snapshot)
            withContext(Dispatchers.Main) {
                if (mode < 0) {
                    isClicking = false
                    OverlayStateHub.setClickingRunning(false)
                    refreshAnchorVisuals()
                    refreshControlBarState()
                    Toast.makeText(
                        this@OverlayService,
                        "启动连点失败 (code=$mode)，请检查 Shizuku 状态",
                        Toast.LENGTH_SHORT
                    ).show()
                    scheduleAutoCollapse(IDLE_COLLAPSE_DELAY_MS)
                } else {
                    engineMode = mode
                    refreshControlBarState()
                    scheduleAutoCollapse(START_COLLAPSE_DELAY_MS)
                }
            }
        }
    }

    private fun stopClickingInternal() {
        if (!isClicking) return
        cancelAutoCollapse()
        isClicking = false
        OverlayStateHub.setClickingRunning(false)
        // Zero WMS updateViewLayout on Stop: only local invalidate()
        refreshAnchorVisuals()
        refreshControlBarState()
        serviceScope.launch(Dispatchers.IO) {
            ShizukuController.stopClicking()
        }
        if (!isBarCollapsed) {
            scheduleAutoCollapse(IDLE_COLLAPSE_DELAY_MS)
        }
    }

    override fun onAddTarget() {
        if (isClicking) {
            stopClickingInternal()
        }
        if (isLocked) {
            applyLockMode(false)
        }
        val metrics = ShizukuController.getDisplayScreenMetrics(this)
        val settings = ClickConfigStore.loadSettings(this)
        val offsetIndex = targets.size % 6
        val offsetPx = (offsetIndex - 2) * 34f * resources.displayMetrics.density
        val newTarget = ClickTarget(
            id = nextTargetId++,
            index = targets.size + 1,
            x = (metrics.widthPx * 0.5f + offsetPx).coerceIn(40f, (metrics.widthPx - 40).coerceAtLeast(40).toFloat()),
            y = (metrics.heightPx * 0.5f + offsetPx * 0.6f).coerceIn(40f, (metrics.heightPx - 40).coerceAtLeast(40).toFloat()),
            intervalMs = settings.defaultIntervalMs,
            pressDurationMs = settings.defaultPressDurationMs,
            microJitter = settings.defaultMicroJitter,
            enabled = true
        )
        targets += newTarget
        selectedTargetId = newTarget.id
        attachSingleAnchor(newTarget)
        persistCurrentTargets()
        refreshAnchorVisuals()
        if (tunePanelView != null) {
            showOrUpdateTunePanel(forceShow = true)
        } else {
            scheduleAutoCollapse(IDLE_COLLAPSE_DELAY_MS)
        }
    }

    override fun onRemoveTarget() {
        if (targets.isEmpty()) return
        if (isClicking) {
            stopClickingInternal()
        }
        val removeId = if (targets.any { it.id == selectedTargetId }) {
            selectedTargetId
        } else {
            targets.last().id
        }
        targets.removeAll { it.id == removeId }
        anchorViews.remove(removeId)?.let { view ->
            try {
                windowManager.removeViewImmediate(view)
            } catch (_: Throwable) {
            }
        }
        anchorLps.remove(removeId)
        selectedTargetId = targets.lastOrNull()?.id ?: -1
        persistCurrentTargets()
        refreshAnchorVisuals()
        if (targets.isEmpty()) {
            dismissTunePanel()
            scheduleAutoCollapse(IDLE_COLLAPSE_DELAY_MS)
        } else if (tunePanelView != null) {
            showOrUpdateTunePanel(forceShow = true)
        } else {
            scheduleAutoCollapse(IDLE_COLLAPSE_DELAY_MS)
        }
    }

    override fun onToggleLockEdit() {
        if (isClicking && isLocked) {
            stopClickingInternal()
        }
        applyLockMode(!isLocked)
        scheduleAutoCollapse(IDLE_COLLAPSE_DELAY_MS)
    }

    override fun onToggleTunePanel() {
        if (tunePanelView != null) {
            dismissTunePanel()
            scheduleAutoCollapse(IDLE_COLLAPSE_DELAY_MS)
        } else {
            cancelAutoCollapse()
            if (isLocked && !isClicking) {
                applyLockMode(false)
            }
            showOrUpdateTunePanel(forceShow = true)
        }
    }

    override fun onCloseOverlay() {
        cancelAutoCollapse()
        stopClickingInternal()
        stopSelf()
    }

    override fun onDragMove(newLpX: Int, newLpY: Int) {
        val bar = controlBarView ?: return
        val lp = controlBarLp ?: return
        val metrics = ShizukuController.getDisplayScreenMetrics(this)
        val w = bar.getCurrentWidthPx()
        val h = bar.getCurrentHeightPx()
        lp.x = newLpX.coerceIn(0, (metrics.widthPx - w).coerceAtLeast(0))
        lp.y = newLpY.coerceIn(0, (metrics.heightPx - h).coerceAtLeast(0))
        dockRight = (lp.x + w / 2) >= (metrics.widthPx / 2)
        bar.setDockRight(dockRight)

    }

    override fun onDragEnd() {
        val bar = controlBarView ?: return
        val lp = controlBarLp ?: return
        val metrics = ShizukuController.getDisplayScreenMetrics(this)
        val w = bar.getCurrentWidthPx()
        dockRight = (lp.x + w / 2) >= (metrics.widthPx / 2)
        bar.setDockRight(dockRight)
        if (isBarCollapsed) {
            lp.x = if (dockRight) (metrics.widthPx - w).coerceAtLeast(0) else 0
            bar.syncWindowPosition(lp.x, lp.y)
            windowManager.updateViewLayout(bar, lp)
            ClickConfigStore.saveControlBarCollapseState(this, true, dockRight)
        } else {
            ClickConfigStore.saveControlBarPosition(this, lp.x, lp.y)
            ClickConfigStore.saveControlBarCollapseState(this, false, dockRight)
            scheduleAutoCollapse(IDLE_COLLAPSE_DELAY_MS)
        }
    }

    private fun onScreenGeometryChanged() {
        val metrics = ShizukuController.getDisplayScreenMetrics(this)
        // Clamp control bar
        val bar = controlBarView
        val barLp = controlBarLp
        if (bar != null && barLp != null) {
            val w = bar.getCurrentWidthPx()
            val h = bar.getCurrentHeightPx()
            barLp.width = w
            barLp.height = h
            barLp.x = if (isBarCollapsed) {
                if (dockRight) (metrics.widthPx - w).coerceAtLeast(0) else 0
            } else {
                barLp.x.coerceIn(0, (metrics.widthPx - w).coerceAtLeast(0))
            }
            barLp.y = barLp.y.coerceIn(0, (metrics.heightPx - h).coerceAtLeast(0))
            bar.syncWindowPosition(barLp.x, barLp.y)
            windowManager.updateViewLayout(bar, barLp)
        }

        // Clamp anchors inside new display bounds
        var anyChanged = false
        for (t in targets) {
            val clampedX = t.x.coerceIn(0f, (metrics.widthPx - 1).coerceAtLeast(1).toFloat())
            val clampedY = t.y.coerceIn(0f, (metrics.heightPx - 1).coerceAtLeast(1).toFloat())
            if (clampedX != t.x || clampedY != t.y) {
                t.x = clampedX
                t.y = clampedY
                anyChanged = true
            }
            val view = anchorViews[t.id]
            val lp = anchorLps[t.id]
            if (view != null && lp != null) {
                lp.x = (t.x - view.halfSizePx).roundToInt()
                lp.y = (t.y - view.halfSizePx).roundToInt()
                view.syncWindowPosition(lp.x, lp.y)
                windowManager.updateViewLayout(view, lp)
            }
        }
        if (anyChanged) {
            ClickConfigStore.saveTargets(this, targets)
        }

        serviceScope.launch(Dispatchers.IO) {
            ShizukuController.updateDisplayMetrics()
            if (isClicking) {
                ShizukuController.updateTargets(targets.map { it.copy() })
            }
        }
    }

    private fun pushTargetsToServiceAsync() {
        val snapshot = targets.map { it.copy() }
        serviceScope.launch(Dispatchers.IO) {
            ShizukuController.updateTargets(snapshot)
        }
    }

data class EngineSnapshot(

    val fallbackReady: Boolean = false,

/**
 * Process-wide Shizuku UserService connection, permission, auto-rebind, and IPC controller.
 */
object ShizukuController {
    private const val TAG = "ShizukuController"
    const val REQUEST_CODE_SHIZUKU = 10042

    private val scope = CoroutineScope(SupervisorJob() + Dispatchers.IO)

    private var appContext: Context? = null
    private var listenersRegistered = false
    private var boundArgs: Shizuku.UserServiceArgs? = null
    @Volatile
    private var service: ITouchMuxService? = null

    private val _connectionState = MutableStateFlow(ShizukuConnectionState.SHIZUKU_NOT_RUNNING)
    val connectionState: StateFlow<ShizukuConnectionState> = _connectionState.asStateFlow()

    private val _engineSnapshot = MutableStateFlow(EngineSnapshot())
    val engineSnapshot: StateFlow<EngineSnapshot> = _engineSnapshot.asStateFlow()

    private val binderReceivedListener = Shizuku.OnBinderReceivedListener {
        Log.i(TAG, "Shizuku binder received.")
        evaluateShizukuState()
    }

    private val binderDeadListener = Shizuku.OnBinderDeadListener {
        Log.w(TAG, "Shizuku binder died; resetting service reference.")
        service = null
        _connectionState.value = ShizukuConnectionState.SHIZUKU_NOT_RUNNING
        _engineSnapshot.value = _engineSnapshot.value.copy(
            engineMode = TouchMuxUserService.MODE_IDLE,
            statusDetail = "Shizuku 服务已断开，等待重新连接"
        )
    }

    private val permissionResultListener = Shizuku.OnRequestPermissionResultListener { requestCode, grantResult ->
        if (requestCode == REQUEST_CODE_SHIZUKU) {
            if (grantResult == PackageManager.PERMISSION_GRANTED) {
                bindUserService()
            } else {
                _connectionState.value = ShizukuConnectionState.NO_PERMISSION
                _engineSnapshot.value = _engineSnapshot.value.copy(
                    statusDetail = "用户拒绝了 Shizuku 授权"
                )
            }
        }
    }

    private val userServiceConnection = object : ServiceConnection {
        override fun onServiceConnected(name: ComponentName?, binder: IBinder?) {
            if (binder == null || !binder.pingBinder()) {
                _connectionState.value = ShizukuConnectionState.ERROR
                _engineSnapshot.value = _engineSnapshot.value.copy(
                    statusDetail = "UserService Binder 无效"
                )
                return
            }
            service = ITouchMuxService.Stub.asInterface(binder)
            _connectionState.value = ShizukuConnectionState.CONNECTED
            try {
                binder.linkToDeath({
                    Log.w(TAG, "TouchMuxUserService remote process died.")
                    service = null
                    if (Shizuku.pingBinder() && hasPermission()) {
                        _connectionState.value = ShizukuConnectionState.BINDING
                        bindUserService()
                    } else {
                        _connectionState.value = ShizukuConnectionState.SHIZUKU_NOT_RUNNING
                    }
                }, 0)
            } catch (_: Throwable) {
            }
            refreshCapabilitiesAsync()
        }

        override fun onServiceDisconnected(name: ComponentName?) {
            service = null
            if (Shizuku.pingBinder() && hasPermission()) {
                _connectionState.value = ShizukuConnectionState.BINDING
            } else {
                _connectionState.value = ShizukuConnectionState.SHIZUKU_NOT_RUNNING
            }
        }
    }

    fun initialize(context: Context) {
        if (appContext == null) {
            appContext = context.applicationContext
        }
        if (!listenersRegistered) {
            Shizuku.addBinderReceivedListenerSticky(binderReceivedListener)
            Shizuku.addBinderDeadListener(binderDeadListener)
            Shizuku.addRequestPermissionResultListener(permissionResultListener)
            listenersRegistered = true
        }
        evaluateShizukuState()
    }

    fun evaluateShizukuState() {
        val binderAlive = try {
            Shizuku.pingBinder()
        } catch (_: Throwable) {
            false
        }
        if (!binderAlive) {
            service = null
            _connectionState.value = ShizukuConnectionState.SHIZUKU_NOT_RUNNING
            _engineSnapshot.value = EngineSnapshot(statusDetail = "请先启动 Shizuku 服务")
            return
        }
        if (!hasPermission()) {
            _connectionState.value = ShizukuConnectionState.NO_PERMISSION

        fun allocPointerId(): Int {
            for (pid in 0 until 10) {
                if (activePointers.none { it.pointerId == pid }) {
                    return pid
                }
            }
            return -1
        }

        fun liftPointer(targetId: Int) {
            val idx = activePointers.indexOfFirst { it.targetId == targetId }
            if (idx < 0) return
            val nowMs = SystemClock.uptimeMillis()
            val action = if (activePointers.size == 1) {
                MotionEvent.ACTION_UP
            } else {
                MotionEvent.ACTION_POINTER_UP or (idx shl MotionEvent.ACTION_POINTER_INDEX_SHIFT)
            }
            injectMotionFrame(downTimeMs, nowMs, action, activePointers)
            activePointers.removeAt(idx)
        }

        fun pressPointer(targetId: Int, x: Float, y: Float): Int {
            val pid = allocPointerId()
            if (pid < 0) return -1
            val nowMs = SystemClock.uptimeMillis()
            if (activePointers.isEmpty()) {
                downTimeMs = nowMs
            }
            activePointers.add(ActivePointer(targetId, pid, x, y))
            val idx = activePointers.size - 1
            val action = if (activePointers.size == 1) {
                MotionEvent.ACTION_DOWN
            } else {
                MotionEvent.ACTION_POINTER_DOWN or (idx shl MotionEvent.ACTION_POINTER_INDEX_SHIFT)
            }
            injectMotionFrame(downTimeMs, nowMs, action, activePointers)
            return pid
        }

        try {
            while (fallbackRunning.get() && clickingActive.get()) {
                val snapshot: List<ClickTargetParceled>
                val dispW: Int
                val dispH: Int
                synchronized(fallbackLock) {
                    snapshot = fallbackTargets
                    dispW = fallbackDisplayW
                    dispH = fallbackDisplayH
                }

                val nowNs = System.nanoTime()
                val enabledIds = HashSet<Int>(snapshot.size)

                for (t in snapshot) {
                    if (!t.enabled) continue
                    enabledIds.add(t.id)
                    val existing = runtimeMap[t.id]
                    if (existing == null) {
                        runtimeMap[t.id] = FallbackTargetState(
                            config = t,
                            pressed = false,
                            pointerId = -1,
                            nextDeadlineNs = nowNs + runtimeMap.size * 500_000L
                        )
                    } else {
                        existing.config = t
                    }
                }

                // Release removed or disabled targets gracefully (ACTION_UP / POINTER_UP, never ACTION_CANCEL)
                val iter = runtimeMap.entries.iterator()
                while (iter.hasNext()) {
                    val entry = iter.next()
                    if (!enabledIds.contains(entry.key)) {
                        if (entry.value.pressed) {
                            liftPointer(entry.key)
                        }
                        iter.remove()
                    }
                }

                var earliestNs = Long.MAX_VALUE
                for (st in runtimeMap.values) {
                    if (st.nextDeadlineNs <= nowNs) {
                        if (!st.pressed) {
                            val jx = if (st.config.microJitter) {
                                (Random.nextFloat() * 8f - 4f)
                            } else 0f
                            val jy = if (st.config.microJitter) {
                                (Random.nextFloat() * 8f - 4f)
                            } else 0f
                            val px = (st.config.x + jx).coerceIn(0f, (dispW - 1).coerceAtLeast(1).toFloat())
                            val py = (st.config.y + jy).coerceIn(0f, (dispH - 1).coerceAtLeast(1).toFloat())
                            val pid = pressPointer(st.config.id, px, py)
                            if (pid >= 0) {
                                st.pressed = true
                                st.pointerId = pid
                                val pressDur = st.config.pressDurationMs.coerceIn(8L, 5000L)
                                st.nextDeadlineNs = nowNs + pressDur * 1_000_000L
                            } else {
                                st.nextDeadlineNs = nowNs + 5_000_000L
                            }
                        } else {
                            liftPointer(st.config.id)
                            st.pressed = false
                            st.pointerId = -1
                            val interval = st.config.intervalMs.coerceIn(5L, 60_000L)
                            st.nextDeadlineNs = nowNs + interval * 1_000_000L
                        }
                    }
                    if (st.nextDeadlineNs < earliestNs) {
                        earliestNs = st.nextDeadlineNs
                    }
                }

                val sleepNs = if (earliestNs == Long.MAX_VALUE) {
                    20_000_000L
                } else {
                    (earliestNs - System.nanoTime()).coerceIn(500_000L, 50_000_000L)
                }
                LockSupport.parkNanos(sleepNs)
            }
        } finally {
            // Gracefully lift any still-pressed virtual pointers without emitting ACTION_CANCEL
            while (activePointers.isNotEmpty()) {
                val lastTargetId = activePointers.last().targetId
                liftPointer(lastTargetId)
            }
        }
    }

    private fun injectMotionFrame(
        downTimeMs: Long,
        eventTimeMs: Long,
        action: Int,
        pointers: List<ActivePointer>
    ) {
        val im = cachedInputManagerObj ?: return
        val method = cachedInjectMethod ?: return
        val count = pointers.size
        if (count == 0) return

        val props = Array(count) { i ->
            MotionEvent.PointerProperties().apply {
                id = pointers[i].pointerId
                toolType = MotionEvent.TOOL_TYPE_FINGER
            }
        }
        val coords = Array(count) { i ->
            MotionEvent.PointerCoords().apply {
                x = pointers[i].x
                y = pointers[i].y
                pressure = 1.0f
                size = 0.08f
            }
        }

        // Use virtual deviceId (-1) instead of physical touchscreen deviceId to avoid colliding
        // directly with physical finger TouchState DOWN/UP transitions inside InputDispatcher.
        val event = MotionEvent.obtain(
            downTimeMs,
            eventTimeMs,
            action,
            count,
            props,
            coords,
            0,
            0,
            1.0f,
            1.0f,
            VIRTUAL_INJECT_DEVICE_ID,
            0,
            InputDevice.SOURCE_TOUCHSCREEN,
            0
        )
        try {
            method.invoke(im, event, INJECT_INPUT_EVENT_MODE_ASYNC)
        } catch (t: Throwable) {
            Log.w(TAG, "injectInputEvent failed: ${t.message}")
        } finally {
            event.recycle()
        }
    }

    companion object {
        private const val TAG = "TouchMuxUserService"
        private const val INJECT_INPUT_EVENT_MODE_ASYNC = 0
        private const val VIRTUAL_INJECT_DEVICE_ID = -1
        private const val NATIVE_STATE_RUNNING_GRABBED = 2

        const val MODE_IDLE = 0
        const val MODE_NATIVE_MUX = 1
        const val MODE_SHIZUKU_FALLBACK = 2
    }
}

    private fun launchTouchSelfCheckFlow() {
        if (!ShizukuController.isServiceConnected()) {
            showSnackbar("请先连接 Shizuku 服务")
            return
        }

        lifecycleScope.launch(Dispatchers.IO) {
            val rc = ShizukuController.startSelfCheckGrab(SELF_CHECK_TIMEOUT_MS)
            withContext(Dispatchers.Main) {
                if (rc != 0) {
                    showSnackbar("无法开启 EVIOCGRAB 自检 (err=$rc)，请确认设备节点权限")
                    return@withContext
                }
                startSelfCheckCountdown()
            }
        }
    }

    private fun startSelfCheckCountdown() {
        selfCheckTimer?.cancel()
        selfCheckRemainingMs = SELF_CHECK_TIMEOUT_MS.toLong()
        selfCheckTimer = object : CountDownTimer(SELF_CHECK_TIMEOUT_MS.toLong(), 100L) {
            override fun onTick(millisUntilFinished: Long) {
                selfCheckRemainingMs = millisUntilFinished.coerceAtLeast(1L)
            }

            override fun onFinish() {
                selfCheckRemainingMs = 0L
                lifecycleScope.launch(Dispatchers.IO) {
                    ShizukuController.cancelSelfCheckAndUngrab()
                }
                ClickConfigStore.markSelfCheckPassed(this@MainActivity, false)
                selfCheckPassed = false
                showSnackbar("5 秒未确认，已自动安全回滚原厂触控")
            }
        }.start()
    }

    private fun confirmTouchSelfCheck() {
        selfCheckTimer?.cancel()
        selfCheckTimer = null
        selfCheckRemainingMs = 0L
        lifecycleScope.launch(Dispatchers.IO) {
            val ok = ShizukuController.confirmSelfCheck()
            withContext(Dispatchers.Main) {
                selfCheckPassed = ok
                if (ok) {
                    showSnackbar("自检通过！零断触原生合流通道已解锁")
                } else {
                    showSnackbar("确认超时，内核已自动回滚至原厂触控")
                }
            }
        }
    }

    private fun rollbackTouchSelfCheck() {
        selfCheckTimer?.cancel()
        selfCheckTimer = null
        selfCheckRemainingMs = 0L
        lifecycleScope.launch(Dispatchers.IO) {
            ShizukuController.cancelSelfCheckAndUngrab()
        }
        ClickConfigStore.markSelfCheckPassed(this, false)
        selfCheckPassed = false
        showSnackbar("已手动安全回滚至原厂触控")
    }

    companion object {
        private const val SELF_CHECK_TIMEOUT_MS = 5000
    }

data class MainUiState(
    val connectionState: ShizukuConnectionState = ShizukuConnectionState.SHIZUKU_NOT_RUNNING,
    val engineSnapshot: EngineSnapshot = EngineSnapshot(),
    val overlayVisible: Boolean = false,
    val clickingRunning: Boolean = false,
    val overlayPermGranted: Boolean = false,
    val notifPermGranted: Boolean = false,
    val showNotifRow: Boolean = false,
    val selfCheckPassed: Boolean = false,
    val intervalMs: Long = ClickTarget.DEFAULT_INTERVAL_MS,
    val pressDurationMs: Long = ClickTarget.DEFAULT_PRESS_DURATION_MS,
    val microJitter: Boolean = true,
    val autoLockOnStart: Boolean = true,
    val autoCollapseBar: Boolean = true,
    val selfCheckRemainingMs: Long = 0L,
    val selfCheckTimeoutMs: Long = 5000L,
    val presets: List<ClickPreset> = emptyList(),
    val activePresetId: String = "",
    val screenWidthPx: Int = 1080,
    val screenHeightPx: Int = 2400
)

data class MainScreenCallbacks(
    val onToggleOverlay: () -> Unit,
    val onGrantOverlay: () -> Unit,
    val onGrantShizuku: () -> Unit,
    val onGrantNotif: () -> Unit,
    val onStartSelfCheck: () -> Unit,
    val onConfirmSelfCheck: () -> Unit,
    val onRollbackSelfCheck: () -> Unit,
    val onReleaseNative: () -> Unit,
    val onIntervalChange: (value: Long, commit: Boolean) -> Unit,
    val onPressChange: (value: Long, commit: Boolean) -> Unit,
    val onJitterChange: (Boolean) -> Unit,
    val onAutoLockChange: (Boolean) -> Unit,
    val onAutoCollapseChange: (Boolean) -> Unit,
    val onApplyDefaultsToCurrentPreset: () -> Unit,
    val onSelectPreset: (String) -> Unit,
    val onCreatePreset: (name: String, copyCurrent: Boolean) -> Unit,
    val onRenamePreset: (presetId: String, newName: String) -> Unit,
    val onDeletePreset: (presetId: String) -> Unit,
    val onSavePresetTarget: (
        targetId: Int?,
        x: Float,
        y: Float,
        intervalMs: Long,
        pressDurationMs: Long,
        microJitter: Boolean,
        enabled: Boolean
    ) -> Unit,
    val onTogglePresetTargetEnabled: (targetId: Int, enabled: Boolean) -> Unit,
    val onDeletePresetTarget: (targetId: Int) -> Unit
)

private enum class NumericEditKind {
    INTERVAL,
    PRESS
}

private data class TargetEditDraft(
    val existingId: Int?,
    val x: Float,
    val y: Float,
    val intervalMs: Long,
    val pressDurationMs: Long,
    val microJitter: Boolean,
    val enabled: Boolean
)

@OptIn(ExperimentalMaterial3Api::class)
@Composable
fun MainScreen(
    state: MainUiState,
    callbacks: MainScreenCallbacks,
    snackbarHostState: SnackbarHostState
) {
    val scrollBehavior = TopAppBarDefaults.pinnedScrollBehavior()
    var editKind by remember { mutableStateOf<NumericEditKind?>(null) }
    var showCreatePresetDialog by remember { mutableStateOf(false) }
    var renamingPreset by remember { mutableStateOf<ClickPreset?>(null) }
    var editingTargetDraft by remember { mutableStateOf<TargetEditDraft?>(null) }

    val activePreset = state.presets.firstOrNull { it.id == state.activePresetId } ?: state.presets.firstOrNull()

    fun openAddTargetDialog() {
        val count = activePreset?.targets?.size ?: 0
        val offset = ((count % 5) - 2) * 80f
        editingTargetDraft = TargetEditDraft(
            existingId = null,
            x = (state.screenWidthPx * 0.5f + offset).coerceIn(40f, (state.screenWidthPx - 40).coerceAtLeast(40).toFloat()),
            y = (state.screenHeightPx * 0.5f + offset * 0.5f).coerceIn(40f, (state.screenHeightPx - 40).coerceAtLeast(40).toFloat()),
            intervalMs = state.intervalMs,
            pressDurationMs = state.pressDurationMs,
            microJitter = state.microJitter,
            enabled = true
        )
    }

    Scaffold(
        modifier = Modifier
            .fillMaxSize()
            .nestedScroll(scrollBehavior.nestedScrollConnection),
        containerColor = MaterialTheme.colorScheme.surface,
        snackbarHost = { SnackbarHost(hostState = snackbarHostState) },
        topBar = {
            CenterAlignedTopAppBar(
                title = {
                    Column(horizontalAlignment = Alignment.CenterHorizontally) {
                        Text(
                            text = "零断触连点器",
                            style = MaterialTheme.typography.titleLarge,
                            fontWeight = FontWeight.SemiBold
                        )
                        Text(
                            text = "TouchMux 硬件合流 · 预设方案 · 极简折叠悬浮胶囊",
                            style = MaterialTheme.typography.labelSmall,
                            color = MaterialTheme.colorScheme.onSurfaceVariant
                        )
                    }
                },
                colors = TopAppBarDefaults.centerAlignedTopAppBarColors(
                    containerColor = MaterialTheme.colorScheme.surface,
                    scrolledContainerColor = MaterialTheme.colorScheme.surfaceVariant.copy(alpha = 0.45f)
                ),
                scrollBehavior = scrollBehavior
            )
        }
    ) { innerPadding ->
        BoxWithConstraints(
            modifier = Modifier
                .fillMaxSize()
                .padding(innerPadding)
                .verticalScroll(rememberScrollState()),
            contentAlignment = Alignment.TopCenter
        ) {
            val isWide = maxWidth >= 700.dp
            Column(
                modifier = Modifier
                    .widthIn(max = if (isWide) 980.dp else 600.dp)
                    .fillMaxWidth()
                    .padding(horizontal = 16.dp, vertical = 12.dp),
                verticalArrangement = Arrangement.spacedBy(14.dp)
            ) {
                if (isWide) {
                    Row(
                        modifier = Modifier.fillMaxWidth(),
                        horizontalArrangement = Arrangement.spacedBy(16.dp),
                        verticalAlignment = Alignment.Top
                    ) {
                        Column(
                            modifier = Modifier.weight(1f),
                            verticalArrangement = Arrangement.spacedBy(14.dp)
                        ) {
                            EngineHeroCard(state = state, onToggleOverlay = callbacks.onToggleOverlay)
                            PermissionCard(state = state, callbacks = callbacks)
                            SelfCheckCard(state = state, callbacks = callbacks)
                        }
                        Column(
                            modifier = Modifier.weight(1f),
                            verticalArrangement = Arrangement.spacedBy(14.dp)
                        ) {
                            PresetSchemesCard(
                                state = state,
                                activePreset = activePreset,
                                onSelectPreset = callbacks.onSelectPreset,
                                onRequestCreatePreset = { showCreatePresetDialog = true },
                                onRequestRenamePreset = { preset -> renamingPreset = preset },
                                onDeletePreset = callbacks.onDeletePreset,
                                onRequestAddTarget = ::openAddTargetDialog,
                                onRequestEditTarget = { target ->
                                    editingTargetDraft = TargetEditDraft(
                                        existingId = target.id,
                                        x = target.x,
                                        y = target.y,
                                        intervalMs = target.intervalMs,
                                        pressDurationMs = target.pressDurationMs,
                                        microJitter = target.microJitter,
                                        enabled = target.enabled
                                    )
                                },
                                onToggleTargetEnabled = callbacks.onTogglePresetTargetEnabled,
                                onDeleteTarget = callbacks.onDeletePresetTarget
                            )
                            GlobalSettingsCard(
                                state = state,
                                callbacks = callbacks,
                                onRequestExactEdit = { editKind = it }
                            )
                        }
                    }
                } else {
                    EngineHeroCard(state = state, onToggleOverlay = callbacks.onToggleOverlay)
                    PresetSchemesCard(
                        state = state,
                        activePreset = activePreset,
                        onSelectPreset = callbacks.onSelectPreset,
                        onRequestCreatePreset = { showCreatePresetDialog = true },
                        onRequestRenamePreset = { preset -> renamingPreset = preset },
                        onDeletePreset = callbacks.onDeletePreset,
                        onRequestAddTarget = ::openAddTargetDialog,
                        onRequestEditTarget = { target ->
                            editingTargetDraft = TargetEditDraft(
                                existingId = target.id,
                                x = target.x,
                                y = target.y,
                                intervalMs = target.intervalMs,
                                pressDurationMs = target.pressDurationMs,
                                microJitter = target.microJitter,
                                enabled = target.enabled
                            )
                        },
                        onToggleTargetEnabled = callbacks.onTogglePresetTargetEnabled,
                        onDeleteTarget = callbacks.onDeletePresetTarget
                    )
                    GlobalSettingsCard(
                        state = state,
                        callbacks = callbacks,
                        onRequestExactEdit = { editKind = it }
                    )
                    PermissionCard(state = state, callbacks = callbacks)
                    SelfCheckCard(state = state, callbacks = callbacks)
                }
                Spacer(modifier = Modifier.height(12.dp))
            }
        }
    }

    if (state.selfCheckRemainingMs > 0L) {
        SelfCheckCountdownDialog(
            remainingMs = state.selfCheckRemainingMs,
            totalMs = state.selfCheckTimeoutMs,
            onConfirm = callbacks.onConfirmSelfCheck,
            onRollback = callbacks.onRollbackSelfCheck
        )
    }

    editKind?.let { kind ->
        ExactMillisInputDialog(
            kind = kind,
            currentValue = if (kind == NumericEditKind.INTERVAL) state.intervalMs else state.pressDurationMs,
            onDismiss = { editKind = null },
            onApply = { value ->
                if (kind == NumericEditKind.INTERVAL) {
                    callbacks.onIntervalChange(value, true)
                } else {
                    callbacks.onPressChange(value, true)
                }
                editKind = null
            }
        )
    }

    if (showCreatePresetDialog) {
        CreatePresetDialog(
            suggestedName = "方案 ${state.presets.size + 1}",
            hasCurrentTargets = (activePreset?.targets?.isNotEmpty() == true),
            onDismiss = { showCreatePresetDialog = false },
            onConfirm = { name, copyCurrent ->
                callbacks.onCreatePreset(name, copyCurrent)
                showCreatePresetDialog = false
            }
        )
    }

    renamingPreset?.let { preset ->
        RenamePresetDialog(
            initialName = preset.name,
            onDismiss = { renamingPreset = null },
            onConfirm = { newName ->
                callbacks.onRenamePreset(preset.id, newName)
                renamingPreset = null
            }
        )
    }

    editingTargetDraft?.let { draft ->
        PresetTargetEditDialog(
            draft = draft,
            screenWidthPx = state.screenWidthPx,
            screenHeightPx = state.screenHeightPx,
            onDismiss = { editingTargetDraft = null },
            onSave = { x, y, intervalMs, pressMs, jitter, enabled ->
                callbacks.onSavePresetTarget(
                    draft.existingId,
                    x,
                    y,
                    intervalMs,
                    pressMs,
                    jitter,
                    enabled
                )
                editingTargetDraft = null
            }
        )
    }
}

@OptIn(ExperimentalLayoutApi::class)
@Composable
private fun EngineHeroCard(
    state: MainUiState,
    onToggleOverlay: () -> Unit
) {
    val conn = state.connectionState
    val snap = state.engineSnapshot
    val nativeCapable = snap.uinputWritable && snap.touchDeviceReadable
    val nativeActive = snap.engineMode == TouchMuxUserService.MODE_NATIVE_MUX
    val fallbackActive = snap.engineMode == TouchMuxUserService.MODE_SHIZUKU_FALLBACK

    val targetContainerColor = when {
        conn != ShizukuConnectionState.CONNECTED -> MaterialTheme.colorScheme.surfaceVariant
        nativeActive || (nativeCapable && state.selfCheckPassed) -> MaterialTheme.colorScheme.primaryContainer
        nativeCapable -> MaterialTheme.colorScheme.tertiaryContainer
        else -> MaterialTheme.colorScheme.secondaryContainer
    }
    val containerColor by animateColorAsState(targetValue = targetContainerColor, label = "heroCardColor")
    val contentColor = when {
        conn != ShizukuConnectionState.CONNECTED -> MaterialTheme.colorScheme.onSurfaceVariant
        nativeActive || (nativeCapable && state.selfCheckPassed) -> MaterialTheme.colorScheme.onPrimaryContainer
        nativeCapable -> MaterialTheme.colorScheme.onTertiaryContainer
        else -> MaterialTheme.colorScheme.onSecondaryContainer
    }

    val headline = when {
        conn == ShizukuConnectionState.SHIZUKU_NOT_RUNNING -> "等待连接 Shizuku 服务"
        conn == ShizukuConnectionState.NO_PERMISSION -> "Shizuku 已运行，待授权"
        conn == ShizukuConnectionState.BINDING -> "正在绑定触控服务..."
        conn == ShizukuConnectionState.ERROR -> "Shizuku 连接异常"
        nativeActive -> "原生 TouchMux 硬件合流激活"
        nativeCapable && state.selfCheckPassed -> "原生 TouchMux 已通过自检，就绪"
        nativeCapable -> "本机支持免 Root TouchMux！建议先完成 5 秒自检"
        fallbackActive -> "Shizuku UID ${snap.uid} 异步注入运行中"
        snap.fallbackReady -> "Shizuku 异步注入通道就绪"
        else -> "引擎待机中"
    }

    val badgeLabel = when {
        state.clickingRunning -> "连点运行中"
        state.overlayVisible -> "悬浮窗已开启"
        nativeCapable && state.selfCheckPassed -> "原生零断触就绪"
        conn == ShizukuConnectionState.CONNECTED -> "服务已连接"
        else -> "未就绪"
    }

    ElevatedCard(
        shape = MaterialTheme.shapes.large,
        colors = CardDefaults.elevatedCardColors(
            containerColor = containerColor,
            contentColor = contentColor
        ),
        modifier = Modifier.fillMaxWidth()
    ) {
        Column(
            modifier = Modifier.padding(18.dp),
            verticalArrangement = Arrangement.spacedBy(12.dp)
        ) {
            Row(
                modifier = Modifier.fillMaxWidth(),
                horizontalArrangement = Arrangement.SpaceBetween,
                verticalAlignment = Alignment.CenterVertically
            ) {
                Row(
                    verticalAlignment = Alignment.CenterVertically,
                    horizontalArrangement = Arrangement.spacedBy(10.dp),
                    modifier = Modifier.weight(1f)
                ) {
                    Surface(
                        shape = CircleShape,
                        color = contentColor.copy(alpha = 0.14f),
                        modifier = Modifier.size(40.dp)
                    ) {
                        Icon(
                            painter = painterResource(R.drawable.ic_touch_app),
                            contentDescription = null,
                            tint = contentColor,
                            modifier = Modifier.padding(8.dp)
                        )
                    }
                    Column {
                        Text(
                            text = "引擎与硬件通道",
                            style = MaterialTheme.typography.labelMedium,
                            color = contentColor.copy(alpha = 0.8f)
                        )
                        Text(
                            text = headline,
                            style = MaterialTheme.typography.titleMedium,
                            fontWeight = FontWeight.Bold
                        )
                    }
                }
                Surface(
                    shape = RoundedCornerShape(50),
                    color = MaterialTheme.colorScheme.surface.copy(alpha = 0.72f),
                    contentColor = MaterialTheme.colorScheme.onSurface
                ) {
                    Text(
                        text = badgeLabel,
                        style = MaterialTheme.typography.labelSmall,
                        fontWeight = FontWeight.SemiBold,
                        modifier = Modifier.padding(horizontal = 10.dp, vertical = 5.dp)
                    )
                }
            }

            if (conn == ShizukuConnectionState.CONNECTED) {
                FlowRow(
                    horizontalArrangement = Arrangement.spacedBy(8.dp),
                    verticalArrangement = Arrangement.spacedBy(6.dp)
                ) {
                    StatusInfoChip(text = "UID ${snap.uid}")
                    StatusInfoChip(
                        text = if (snap.uinputWritable) "/dev/uinput 可写" else "/dev/uinput 受限"
                    )
                    StatusInfoChip(
                        text = if (snap.touchDeviceReadable) "物理节点可读" else "物理节点受限"
                    )
                    StatusInfoChip(
                        text = if (snap.fallbackReady) "Fallback 可用" else "Fallback 不可用"
                    )
                }

                if (snap.touchDeviceInfo.isNotBlank() || snap.nativeProbeError.isNotBlank()) {
                    Text(
                        text = buildString {
                            if (snap.touchDeviceInfo.isNotBlank()) {
                                append("触控设备: ").append(snap.touchDeviceInfo)
                            }
                            if (snap.nativeProbeError.isNotBlank()) {
                                if (isNotEmpty()) append("\n")
                                append("诊断: ").append(snap.nativeProbeError)
                            }
                        },
                        style = MaterialTheme.typography.bodySmall,
                        color = contentColor.copy(alpha = 0.82f)
                    )
                }
            } else {
                Text(
                    text = snap.statusDetail,
                    style = MaterialTheme.typography.bodySmall,
                    color = contentColor.copy(alpha = 0.85f)
                )
            }

            Button(
                onClick = onToggleOverlay,
                shape = RoundedCornerShape(28.dp),
                contentPadding = PaddingValues(vertical = 15.dp, horizontal = 20.dp),
                colors = if (state.overlayVisible) {
                    ButtonDefaults.buttonColors(
                        containerColor = MaterialTheme.colorScheme.error,
                        contentColor = MaterialTheme.colorScheme.onError
                    )
                } else {
                    ButtonDefaults.buttonColors(
                        containerColor = MaterialTheme.colorScheme.primary,
                        contentColor = MaterialTheme.colorScheme.onPrimary
                    )
                },
                modifier = Modifier.fillMaxWidth()
            ) {
                Icon(
                    imageVector = if (state.overlayVisible) Icons.Rounded.Close else Icons.Rounded.PlayArrow,
                    contentDescription = null,
                    modifier = Modifier.size(20.dp)
                )
                Spacer(modifier = Modifier.width(8.dp))
                Text(
                    text = if (state.overlayVisible) "关闭游戏悬浮控制条" else "开启游戏悬浮控制条",
                    style = MaterialTheme.typography.titleSmall,
                    fontWeight = FontWeight.SemiBold
                )
            }
        }
    }
}

@Composable
private fun StatusInfoChip(text: String) {
    AssistChip(
        onClick = {},
        label = {
            Text(
                text = text,
                style = MaterialTheme.typography.labelSmall,
                maxLines = 1,
                overflow = TextOverflow.Ellipsis
            )
        },
        colors = AssistChipDefaults.assistChipColors(
            containerColor = MaterialTheme.colorScheme.surface.copy(alpha = 0.65f),
            labelColor = MaterialTheme.colorScheme.onSurface
        ),
        border = null,
        modifier = Modifier.height(28.dp)
    )
}

@Composable
private fun PermissionCard(
    state: MainUiState,
    callbacks: MainScreenCallbacks
) {
    val conn = state.connectionState
    val shizukuSubtitle = when (conn) {
        ShizukuConnectionState.CONNECTED -> "已授权并绑定特权触控服务"
        ShizukuConnectionState.NO_PERMISSION -> "待授权，点击右侧授权"
        ShizukuConnectionState.BINDING -> "正在绑定 UserService..."
        ShizukuConnectionState.SHIZUKU_NOT_RUNNING -> "Shizuku 未启动，点击去启动或刷新"
        ShizukuConnectionState.ERROR -> "连接异常，请重启 Shizuku 后重试"
    }
    val shizukuAction = when (conn) {
        ShizukuConnectionState.CONNECTED -> "重测"
        ShizukuConnectionState.NO_PERMISSION -> "授权"
        else -> "启动/刷新"
    }

    OutlinedCard(
        shape = MaterialTheme.shapes.large,
        modifier = Modifier.fillMaxWidth()
    ) {
        Column(
            modifier = Modifier.padding(16.dp),
            verticalArrangement = Arrangement.spacedBy(12.dp)
        ) {
            Text(
                text = "基础权限向导",
                style = MaterialTheme.typography.titleMedium,
                fontWeight = FontWeight.SemiBold
            )

            PermissionActionRow(
                iconRes = R.drawable.ic_layers,
                title = "1. 悬浮窗显示权限",
                subtitle = if (state.overlayPermGranted) "已开启（游戏内可显示控制胶囊与锚点）" else "未授权（悬浮窗必需）",
                granted = state.overlayPermGranted,
                buttonText = if (state.overlayPermGranted) "已开启" else "去开启",
                onClick = callbacks.onGrantOverlay
            )

            HorizontalDivider(color = MaterialTheme.colorScheme.outlineVariant.copy(alpha = 0.5f))

            PermissionActionRow(
                iconRes = R.drawable.ic_touch_app,
                title = "2. Shizuku 跨进程触控权限",
                subtitle = shizukuSubtitle,
                granted = conn == ShizukuConnectionState.CONNECTED,
                buttonText = shizukuAction,
                onClick = callbacks.onGrantShizuku
            )

            if (state.showNotifRow) {
                HorizontalDivider(color = MaterialTheme.colorScheme.outlineVariant.copy(alpha = 0.5f))
                PermissionActionRow(
                    iconRes = R.drawable.ic_notifications,
                    title = "3. 通知权限（前台防杀 & 急停）",
                    subtitle = if (state.notifPermGranted) "已授权" else "未授权（推荐开启防止切后台被杀）",
                    granted = state.notifPermGranted,
                    buttonText = if (state.notifPermGranted) "已授权" else "去授权",
                    buttonEnabled = !state.notifPermGranted,
                    onClick = callbacks.onGrantNotif
                )
            }
        }
    }
}

@Composable
private fun PermissionActionRow(
    iconRes: Int,
    title: String,
    subtitle: String,
    granted: Boolean,
    buttonText: String,
    buttonEnabled: Boolean = true,
    onClick: () -> Unit
) {
    Row(
        modifier = Modifier.fillMaxWidth(),
        verticalAlignment = Alignment.CenterVertically,
        horizontalArrangement = Arrangement.spacedBy(12.dp)
    ) {
        Surface(
            shape = CircleShape,
            color = if (granted) {
                MaterialTheme.colorScheme.primaryContainer
            } else {
                MaterialTheme.colorScheme.surfaceVariant
            },
            modifier = Modifier.size(36.dp)
        ) {
            Icon(
                painter = painterResource(iconRes),
                contentDescription = null,
                tint = if (granted) {
                    MaterialTheme.colorScheme.onPrimaryContainer
                } else {
                    MaterialTheme.colorScheme.onSurfaceVariant
                },
                modifier = Modifier.padding(8.dp)
            )
        }
        Column(modifier = Modifier.weight(1f)) {
            Text(
                text = title,
                style = MaterialTheme.typography.bodyMedium,
                fontWeight = FontWeight.Medium
            )
            Text(
                text = subtitle,
                style = MaterialTheme.typography.bodySmall,
                color = if (granted) {
                    MaterialTheme.colorScheme.primary
                } else {
                    MaterialTheme.colorScheme.onSurfaceVariant
                }
            )
        }
        FilledTonalButton(
            onClick = onClick,
            enabled = buttonEnabled,
            contentPadding = PaddingValues(horizontal = 14.dp, vertical = 6.dp)
        ) {
            if (granted && !buttonEnabled) {
                Icon(
                    imageVector = Icons.Rounded.Check,
                    contentDescription = null,
                    modifier = Modifier.size(16.dp)
                )
                Spacer(modifier = Modifier.width(4.dp))
            }
            Text(text = buttonText, style = MaterialTheme.typography.labelLarge)
        }
    }
}

@Composable
private fun SelfCheckCard(
    state: MainUiState,
    callbacks: MainScreenCallbacks
) {
    val conn = state.connectionState
    val snap = state.engineSnapshot
    val nativeCapable = snap.uinputWritable && snap.touchDeviceReadable
    val selfCheckEnabled = conn == ShizukuConnectionState.CONNECTED && nativeCapable

    val hintText = when {
        conn == ShizukuConnectionState.CONNECTED && !nativeCapable ->

@Composable
private fun ExactMillisInputDialog(
    kind: NumericEditKind,
    currentValue: Long,
    onDismiss: () -> Unit,
    onApply: (Long) -> Unit
) {
    val minVal = if (kind == NumericEditKind.INTERVAL) ClickTarget.MIN_INTERVAL_MS else ClickTarget.MIN_PRESS_MS
    val maxVal = if (kind == NumericEditKind.INTERVAL) ClickTarget.MAX_INTERVAL_MS else ClickTarget.MAX_PRESS_MS
    val title = if (kind == NumericEditKind.INTERVAL) "设置默认点击间隔 (ms)" else "设置默认按压时长 (ms)"
    var inputText by remember(currentValue) { mutableStateOf(currentValue.toString()) }

    AlertDialog(
        onDismissRequest = onDismiss,
        title = { Text(text = title) },
        text = {
            OutlinedTextField(
                value = inputText,
                onValueChange = { raw ->
                    inputText = raw.filter { it.isDigit() }.take(6)
                },
                singleLine = true,
                label = { Text(text = "范围 $minVal ~ $maxVal ms") },
                suffix = { Text(text = "ms") },
                keyboardOptions = KeyboardOptions(keyboardType = KeyboardType.Number),
                modifier = Modifier.fillMaxWidth()
            )
        },
        confirmButton = {
            Button(onClick = {
                val parsed = inputText.toLongOrNull() ?: currentValue
                onApply(parsed.coerceIn(minVal, maxVal))
            }) {
                Text(text = "确定")
            }
        },
        dismissButton = {
            TextButton(onClick = onDismiss) {
                Text(text = "取消")
            }
        }
    )
}

        title = { Text(text = if (draft.existingId == null) "添加预设连点位" else "编辑连点位") },
        text = {
            Column(verticalArrangement = Arrangement.spacedBy(10.dp)) {
                Text(
                    text = "屏幕参考分辨率：$screenWidthPx × $screenHeightPx px（也可开启悬浮窗直接拖动锚点并自动保存）",
                    style = MaterialTheme.typography.bodySmall,
                    color = MaterialTheme.colorScheme.onSurfaceVariant
                )
                Row(horizontalArrangement = Arrangement.spacedBy(8.dp)) {
                    OutlinedTextField(
                        value = xText,
                        onValueChange = { raw -> xText = raw.filter { it.isDigit() }.take(5) },
                        singleLine = true,
                        label = { Text(text = "X 坐标") },
                        keyboardOptions = KeyboardOptions(keyboardType = KeyboardType.Number),
                        modifier = Modifier.weight(1f)
                    )
                    OutlinedTextField(
                        value = yText,
                        onValueChange = { raw -> yText = raw.filter { it.isDigit() }.take(5) },
                        singleLine = true,
                        label = { Text(text = "Y 坐标") },
                        keyboardOptions = KeyboardOptions(keyboardType = KeyboardType.Number),
                        modifier = Modifier.weight(1f)
                    )
                }
                Row(horizontalArrangement = Arrangement.spacedBy(8.dp)) {
                    OutlinedTextField(
                        value = intervalText,
                        onValueChange = { raw -> intervalText = raw.filter { it.isDigit() }.take(5) },
                        singleLine = true,
                        label = { Text(text = "间隔 (ms)") },
                        keyboardOptions = KeyboardOptions(keyboardType = KeyboardType.Number),
                        modifier = Modifier.weight(1f)
                    )
                    OutlinedTextField(
                        value = pressText,
                        onValueChange = { raw -> pressText = raw.filter { it.isDigit() }.take(4) },
                        singleLine = true,
                        label = { Text(text = "按压 (ms)") },
                        keyboardOptions = KeyboardOptions(keyboardType = KeyboardType.Number),
                        modifier = Modifier.weight(1f)
                    )
                }
                Row(
                    modifier = Modifier.fillMaxWidth(),
                    verticalAlignment = Alignment.CenterVertically,
                    horizontalArrangement = Arrangement.SpaceBetween
                ) {
                    Text(text = "随机微抖动 (±4px)", style = MaterialTheme.typography.bodyMedium)
                    Switch(checked = jitter, onCheckedChange = { jitter = it })
                }
                Row(
                    modifier = Modifier.fillMaxWidth(),
                    verticalAlignment = Alignment.CenterVertically,
                    horizontalArrangement = Arrangement.SpaceBetween
                ) {
                    Text(text = "启用该连点位", style = MaterialTheme.typography.bodyMedium)
                    Switch(checked = enabled, onCheckedChange = { enabled = it })
                }
            }
        },
        confirmButton = {
            Button(onClick = {
                val parsedX = xText.toFloatOrNull() ?: draft.x
                val parsedY = yText.toFloatOrNull() ?: draft.y
                val parsedInterval = intervalText.toLongOrNull() ?: draft.intervalMs
                val parsedPress = pressText.toLongOrNull() ?: draft.pressDurationMs
                onSave(parsedX, parsedY, parsedInterval, parsedPress, jitter, enabled)
            }) {
                Text(text = "确定")
            }
        },
        dismissButton = {
            TextButton(onClick = onDismiss) {
                Text(text = "取消")
            }
        }
    )
}
